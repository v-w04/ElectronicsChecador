function getFechaActualMexico() {
  return Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd');
}

function getFechaHoraActualMexico() {
  return Utilities.formatDate(new Date(), TIMEZONE, 'dd/MM/yyyy HH:mm:ss');
}

function formatearFechaMexico(fecha, formato) {
  return Utilities.formatDate(new Date(fecha), TIMEZONE, formato || 'yyyy-MM-dd');
}

function formatearFecha(fecha) {
  const year = fecha.getFullYear();
  const month = String(fecha.getMonth() + 1).padStart(2, '0');
  const day = String(fecha.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}

/**
 * Convierte hora string "HH:mm" a minutos totales.
 * Acepta Date objects, strings con formato HH:mm, y retorna 0 si es inválido.
 */
function convertirHoraAMinutos(horaStr) {
  if (!horaStr) return 0;
  const str = horaStr.toString().trim();
  const match = str.match(/(\d{1,2}):(\d{2})/);
  if (match) {
    return parseInt(match[1]) * 60 + parseInt(match[2]);
  }
  return 0;
}

/**
 * Formatea un valor de celda (Date o string) a "HH:mm".
 */
function formatearHora(valorCelda) {
  if (!valorCelda) return '00:00';
  if (valorCelda instanceof Date) {
    return `${String(valorCelda.getHours()).padStart(2, '0')}:${String(valorCelda.getMinutes()).padStart(2, '0')}`;
  }
  const str = valorCelda.toString();
  const match = str.match(/(\d{1,2}):(\d{2})/);
  if (match) return `${String(match[1]).padStart(2, '0')}:${match[2]}`;
  return '00:00';
}

function calcularDiferenciaMinutos(horaInicio, horaFin) {
  return convertirHoraAMinutos(horaFin) - convertirHoraAMinutos(horaInicio);
}

function esDiaLaboral(fecha) {
  const date = fecha instanceof Date ? fecha : new Date(fecha);
  const dia = date.getDay();
  return dia >= 1 && dia <= 5;
}

function obtenerDiaSemana(fecha) {
  return ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'][fecha.getDay()];
}

}

/**
 * Elimina todos los triggers de enviarReporteDiario
 * @returns {number} Cantidad de triggers eliminados
 */
function eliminarTriggersWhatsApp() {
  const triggers = ScriptApp.getProjectTriggers();
  let eliminados = 0;
  
  triggers.forEach(trigger => {
    if (trigger.getHandlerFunction() === 'enviarReporteDiario') {
      ScriptApp.deleteTrigger(trigger);
      eliminados++;
      Logger.log('🗑️ Trigger eliminado: ' + trigger.getUniqueId());
    }
  });
  
  Logger.log(`✅ Total triggers eliminados: ${eliminados}`);
  return eliminados;
}

// Helper genérico: elimina todos los triggers que apuntan a una función
function _eliminarTriggersPorFuncion(nombreFuncion) {
  const triggers = ScriptApp.getProjectTriggers();
  let eliminados = 0;
  triggers.forEach(t => {
    if (t.getHandlerFunction() === nombreFuncion) {
      ScriptApp.deleteTrigger(t);
      eliminados++;
    }
  });
  Logger.log('🗑️ ' + eliminados + ' trigger(s) eliminados de ' + nombreFuncion);
  return eliminados;
}

// Handlers — Control de Asistencia + WhatsApp encadenados.
// Cuando se dispara el trigger, primero procesa asistencias (puede tardar
// ~10 minutos). Al terminar SIN error, automáticamente envía el reporte de
// WhatsApp con los datos ya frescos en METRICAS_DIARIAS.
// Si el Control falla, el WhatsApp NO se envía (evita reporte con datos viejos).

function _ejecutarControlYReporte(etiqueta, fechaForzada) {
  Logger.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  Logger.log('🚀 TRIGGER ' + etiqueta + ' — Control de Asistencia + WhatsApp');
  if (fechaForzada) Logger.log('🧪 MODO TEST — reportando datos del: ' + fechaForzada);
  Logger.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  const inicio = new Date();
  try {
    Logger.log('📊 [1/2] Iniciando Control de Asistencia...');
    procesarAsistenciasCompleto();
    const segs = Math.round((new Date() - inicio) / 1000);
    Logger.log('✅ [1/2] Control de Asistencia completado en ' + segs + 's');

    // ── VERIFICACIÓN: confirmar que METRICAS_DIARIAS tiene datos de la
    // fecha esperada (hoy real, o la fecha forzada si es test).
    const sheetMetricas = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('METRICAS_DIARIAS');
    if (!sheetMetricas) throw new Error('METRICAS_DIARIAS no existe después del Control');
    const fechaObjetivo = fechaForzada || Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd');
    const lastRow = sheetMetricas.getLastRow();
    if (lastRow < 2) throw new Error('METRICAS_DIARIAS está vacía después del Control');

    const datos = sheetMetricas.getRange(2, 3, lastRow - 1, 1).getValues();
    let filasObjetivo = 0;
    datos.forEach(r => {
      const f = r[0];
      const fStr = f instanceof Date ? Utilities.formatDate(f, TIMEZONE, 'yyyy-MM-dd') : String(f).substring(0, 10);
      if (fStr === fechaObjetivo) filasObjetivo++;
    });
    Logger.log('🔍 Verificación: ' + filasObjetivo + ' filas en METRICAS_DIARIAS con fecha ' + fechaObjetivo);
    if (filasObjetivo === 0) {
      throw new Error('No hay filas en METRICAS_DIARIAS con fecha ' + fechaObjetivo);
    }

    Logger.log('📱 [2/2] Iniciando envío de reporte WhatsApp...');
    enviarReporteDiario(fechaForzada);  // ← pasa fechaForzada si está en modo test
    Logger.log('✅ [2/2] Reporte WhatsApp enviado');

    const total = Math.round((new Date() - inicio) / 1000);
    Logger.log('🎉 TRIGGER ' + etiqueta + ' COMPLETO en ' + total + 's');
  } catch (e) {
    Logger.log('❌ ERROR en trigger ' + etiqueta + ': ' + e.message);
    Logger.log('   El WhatsApp NO se envió para evitar reportes con datos incorrectos.');
    throw e;
  }
}

  _eliminarTriggersPorFuncion('ejecutarTriggerPrueba');

  // Si estamos en fin de semana, usar el viernes anterior como referencia
  const ahora = new Date();
  const dia = ahora.getDay();
  let fechaForzada = null;
  if (dia === 0 || dia === 6) {
    const viernes = new Date(ahora);
    viernes.setDate(ahora.getDate() - (dia === 0 ? 2 : 1)); // domingo→viernes-2, sábado→viernes-1

    Logger.log('🗓️ Fin de semana detectado — el reporte usará datos del viernes ' + fechaForzada);
  }

  _ejecutarControlYReporte('PRUEBA', fechaForzada);
}

function _activarTriggerPruebaEnHora(hora, minuto) {
  const ui = SpreadsheetApp.getUi();
  _eliminarTriggersPorFuncion('ejecutarTriggerPrueba');

  // Construir Date para HOY a la hora pedida en timezone México
  const ahora = new Date();
  const hoyMex = Utilities.formatDate(ahora, 'America/Mexico_City', 'yyyy-MM-dd');
  const partes = hoyMex.split('-').map(Number);

  // Crear Date en el timezone del servidor que represente esa hora en CDMX.
  // Usamos un truco: formatear con offset -06:00 (CDMX no usa DST desde 2022)
  const fechaStr = hoyMex + 'T' +
                   String(hora).padStart(2,'0') + ':' +
                   String(minuto).padStart(2,'0') + ':00-06:00';
  const fechaDisparo = new Date(fechaStr);

  if (fechaDisparo <= ahora) {
    ui.alert('⚠️ Hora ya pasó',
      'Ya son más de las ' + hora + ':' + String(minuto).padStart(2,'0') +
      ' AM hora de México. Elige una hora futura.',
      ui.ButtonSet.OK);
    return;
  }

  try {
    ScriptApp.newTrigger('ejecutarTriggerPrueba')
      .timeBased().at(fechaDisparo).create();
    const horaMexCDMX = Utilities.formatDate(fechaDisparo, 'America/Mexico_City', 'HH:mm');
    Logger.log('✅ Trigger PRUEBA activado para hoy ' + horaMexCDMX + ' (CDMX)');
    ui.alert('🧪 Trigger de PRUEBA activado',
      'Hoy a las ' + horaMexCDMX + ' (hora de México) se ejecutará:\n\n' +
      '1️⃣ Control de Asistencia completo (~10 min)\n' +
      '2️⃣ Verificación de datos en METRICAS_DIARIAS\n' +
      '3️⃣ Reporte WhatsApp (solo si los datos están bien)\n\n' +
      'Este trigger SE AUTOELIMINA al terminar, así que no se vuelve a disparar.\n\n' +
      '⏱️ Para ver el progreso en vivo: Apps Script → Ejecuciones',
      ui.ButtonSet.OK);
  } catch (error) {
    ui.alert('❌ Error', 'No se pudo activar: ' + error.message, ui.ButtonSet.OK);
  }
}

function agregarLog(sheet, etapa, mensaje, estado) {
  const timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'HH:mm:ss.SSS');
  sheet.appendRow([timestamp, etapa, mensaje, estado]);
  SpreadsheetApp.flush();
}

function formatearHeaderLogs(sheet) {
  const range = sheet.getRange(1, 1, 1, 4);
  range.setBackground('#1a237e').setFontColor('#ffffff').setFontWeight('bold').setHorizontalAlignment('center');
  sheet.setColumnWidth(1, 120);
  sheet.setColumnWidth(2, 150);
  sheet.setColumnWidth(3, 500);
  sheet.setColumnWidth(4, 100);
  sheet.setFrozenRows(1);
}

function obtenerUltimosLogs(cantidadLineas) {
  try {
    const logsSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('LOGS_PROCESO');
    if (!logsSheet) return { error: true, message: 'Hoja LOGS_PROCESO no encontrada', logs: [] };

    const lastRow = logsSheet.getLastRow();
    if (lastRow <= 1) return { error: false, logs: [], lastRow: 1 };

    const numLineas = Math.min(cantidadLineas || 10, lastRow - 1);
    const startRow = Math.max(2, lastRow - numLineas + 1);
    const values = logsSheet.getRange(startRow, 1, lastRow - startRow + 1, 2).getValues();

    return {
      error: false,
      logs: values.map(row => ({ timestamp: row[0], message: row[1] })).filter(log => log.message),
      lastRow: lastRow
    };
  } catch (error) {
    return { error: true, message: error.toString(), logs: [] };
  }
}

function procesarAsistenciasCompleto() {
  let desdeWeb = false;
  try { SpreadsheetApp.getUi(); } catch (e) { desdeWeb = true; }

  const ui = desdeWeb ? null : SpreadsheetApp.getUi();

  if (!desdeWeb) {
    if (ui.alert('⚠️ Confirmación', '¿Deseas ejecutar el control de asistencia completo?', ui.ButtonSet.YES_NO) !== ui.Button.YES) {
      return { error: true, message: 'Proceso cancelado por el usuario' };
    }
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheetLogs = ss.getSheetByName('LOGS_PROCESO') || ss.insertSheet('LOGS_PROCESO');
  sheetLogs.clear();
  sheetLogs.appendRow(['⏰ Timestamp', '📊 Etapa', '📝 Mensaje', '✅ Estado']);
  formatearHeaderLogs(sheetLogs);

  const logStart = new Date();

  try {
    agregarLog(sheetLogs, 'INICIO', '🚀 Iniciando procesamiento...', 'INFO');

    agregarLog(sheetLogs, 'SETUP', '🆕 Verificando hojas...', 'INFO');
    crearHojasNecesarias();
    agregarLog(sheetLogs, 'SETUP', '✅ Hojas verificadas', 'OK');

    agregarLog(sheetLogs, 'VALIDACIÓN', '🔍 Validando CONFIG_TURNOS...', 'INFO');
    const sheetConfigTurnos = ss.getSheetByName(CONFIG.HOJAS.CONFIG_TURNOS);
    if (!sheetConfigTurnos || sheetConfigTurnos.getLastRow() < 2) {
      const msg = !sheetConfigTurnos ? 'La hoja CONFIG_TURNOS no existe.' : 'CONFIG_TURNOS está vacía.';
      agregarLog(sheetLogs, 'ERROR', `❌ ${msg}`, 'ERROR');
      if (!desdeWeb) ui.alert('❌ Error de Configuración', msg, ui.ButtonSet.OK);
      return { error: true, message: msg };
    }
    agregarLog(sheetLogs, 'VALIDACIÓN', `✅ CONFIG_TURNOS válida: ${sheetConfigTurnos.getLastRow() - 1} turnos`, 'OK');

    agregarLog(sheetLogs, 'PASO 0/7', '🔄 Creando RAW_COMBINADO...', 'INFO');
    const totalCombinados = crearRawCombinado();
    agregarLog(sheetLogs, 'PASO 0/7', `✅ RAW_COMBINADO: ${totalCombinados} registros`, 'OK');

    agregarLog(sheetLogs, 'PASO 0B/7', '🚛 Fusionando CHECADOR_CHOFERES...', 'INFO');
const totalChofersFusionados = fusionarChecadasChoferes();
agregarLog(sheetLogs, 'PASO 0B/7', `✅ Choferes fusionados: ${totalChofersFusionados}`, 'OK');

    agregarLog(sheetLogs, 'PASO 1/7', '📋 Leyendo datos...', 'INFO');
    const resultadoLimpieza = limpiarDatosRAW();
    const datosLimpios = resultadoLimpieza.datos || resultadoLimpieza;
    agregarLog(sheetLogs, 'PASO 1/7', `✅ ${datosLimpios.length} registros válidos`, 'OK');

    if (datosLimpios.length === 0) {
      agregarLog(sheetLogs, 'ADVERTENCIA', '⚠️ Sin datos válidos en RAW', 'WARNING');
      ['METRICAS_DIARIAS', 'RESUMEN_MENSUAL', 'ALERTAS'].forEach(limpiarSheetMetricas);
      if (!desdeWeb && ui) ui.alert('⚠️ Sin Datos', 'No se encontraron datos válidos. Las métricas han sido limpiadas.', ui.ButtonSet.OK);
      return { success: true, message: 'Sin datos para procesar', registros: 0 };
    }

    agregarLog(sheetLogs, 'PASO 2/7', '📁 Agrupando por empleado y fecha...', 'INFO');
    const datosAgrupados = agruparPorEmpleadoFecha(datosLimpios);
    agregarLog(sheetLogs, 'PASO 2/7', `✅ ${Object.keys(datosAgrupados).length} grupos`, 'OK');

    agregarLog(sheetLogs, 'PASO 3/7', '🔍 Clasificando checadas...', 'INFO');
    const checadasClasificadas = clasificarTodasLasChecadas(datosAgrupados);
    agregarLog(sheetLogs, 'PASO 3/7', `✅ ${checadasClasificadas.length} registros clasificados`, 'OK');

    agregarLog(sheetLogs, 'PASO 4/7', '📊 Calculando métricas...', 'INFO');
    const metricas = calcularMetricasDiarias(checadasClasificadas);
    agregarLog(sheetLogs, 'PASO 4/7', `✅ ${metricas.length} métricas`, 'OK');

    agregarLog(sheetLogs, 'PASO 4B/7', '👥 Completando empleados ausentes (sin checada)...', 'INFO');
    const ausentesAgregados = completarEmpleadosAusentes(metricas);
    metricas.push(...ausentesAgregados);
    agregarLog(sheetLogs, 'PASO 4B/7', `✅ ${ausentesAgregados.length} faltas detectadas por ausencia total`, 'OK');

    agregarLog(sheetLogs, 'PASO 5/7', '⚠️ Generando alertas...', 'INFO');
    const alertas = generarAlertas(metricas);
    agregarLog(sheetLogs, 'PASO 5/7', `✅ ${alertas.length} alertas`, alertas.length > 0 ? 'WARNING' : 'OK');

    agregarLog(sheetLogs, 'PASO 6/7', '📈 Generando resumen mensual...', 'INFO');
    const resumen = generarResumenMensual(metricas);
    agregarLog(sheetLogs, 'PASO 6/7', `✅ ${resumen.length} empleados`, 'OK');

    agregarLog(sheetLogs, 'PASO 7/7', '💾 Escribiendo resultados...', 'INFO');
    escribirClasificacionVisual(checadasClasificadas);
    escribirProcesado(checadasClasificadas);
    escribirMetricas(metricas);
    escribirAlertas(alertas);
    escribirResumen(resumen);
    agregarLog(sheetLogs, 'PASO 7/7', '✅ 5 hojas escritas', 'OK');

    const tiempoTotal = ((new Date() - logStart) / 1000).toFixed(2);
    agregarLog(sheetLogs, 'FIN', `🎉 Completado en ${tiempoTotal}s`, 'SUCCESS');

    ss.setActiveSheet(sheetLogs);
    ss.moveActiveSheet(ss.getNumSheets());

    if (!desdeWeb && ui) {
      ui.alert('✅ ¡Proceso Completado!',
        `Registros: ${datosLimpios.length}\nMétricas: ${metricas.length}\nAlertas: ${alertas.length}\nTiempo: ${tiempoTotal}s`,
        ui.ButtonSet.OK);
      if (ui.alert('📊 Dashboard', '¿Deseas abrir el Dashboard?', ui.ButtonSet.YES_NO) === ui.Button.YES) {
        abrirDashboard();
      }
    }

    return {
      error: false,
      message: 'Proceso completado exitosamente',
      registros: datosLimpios.length,
      metricas: metricas.length,
      alertas: alertas.length,
      tiempo: tiempoTotal,
      timestamp: new Date().toISOString()
    };

  } catch (error) {
    Logger.log('ERROR en procesarAsistenciasCompleto: ' + error);
    agregarLog(sheetLogs, 'ERROR', '❌ ' + error.message, 'ERROR');
    agregarLog(sheetLogs, 'ERROR', 'Stack: ' + error.stack, 'ERROR');
    if (!desdeWeb && ui) ui.alert('❌ Error', error.message + '\n\nRevisa LOGS_PROCESO.', ui.ButtonSet.OK);
    return { error: true, message: error.message, stack: error.stack, timestamp: new Date().toISOString() };
  }
}

function crearRawCombinado() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetRaw = ss.getSheetByName('RAW');
  const sheetRawAdolfo = ss.getSheetByName('RAW_ADOLFO');

  // Eliminar RAW_COMBINADO anterior
  const sheetAnterior = ss.getSheetByName('RAW_COMBINADO');
  if (sheetAnterior) ss.deleteSheet(sheetAnterior);

  let sheetCombinado;
  if (sheetRaw) {
    // Crear copia de RAW para heredar formato y headers
    sheetCombinado = sheetRaw.copyTo(ss);
    sheetCombinado.setName('RAW_COMBINADO');
    // Limpiar contenido (mantener headers)
    const ultimaFila = sheetCombinado.getLastRow();
    if (ultimaFila > 1) {
      sheetCombinado.getRange(2, 1, ultimaFila - 1, sheetCombinado.getLastColumn()).clearContent();
    }
  } else {
    // ⭐ Caso: ya no existe RAW (solo se usa CHECADOR_CHOFERES). Crear hoja
    // vacía con los headers estándar para que el resto del pipeline funcione.
    Logger.log('ℹ️ Hoja RAW no existe — RAW_COMBINADO se creará vacío para choferes');
    sheetCombinado = ss.insertSheet('RAW_COMBINADO');
    const headersEstandar = [
      'ID Usuario', 'Nombre', 'Fecha', 'Turno',
      '', '', '', '',
      'Entrada', 'Salida'
    ];
    sheetCombinado.getRange(1, 1, 1, headersEstandar.length).setValues([headersEstandar]);
    sheetCombinado.getRange(1, 1, 1, headersEstandar.length)
      .setBackground('#1a237e').setFontColor('#ffffff').setFontWeight('bold');
  }

  // PASO 1: Procesar RAW_ADOLFO y construir Set de nombres únicos
  const nombresEnAdolfo = new Set();
  const datosAdolfoFinal = [];

  if (sheetRawAdolfo) {
    const datosAdolfo = sheetRawAdolfo.getDataRange().getValues();
    const formatosAdolfo = sheetRawAdolfo.getDataRange().getNumberFormats();

    for (let i = 1; i < datosAdolfo.length; i++) {
      const nombre = datosAdolfo[i][1];
      if (!nombre || nombre.toString().toUpperCase() === 'SIN USUARIO') continue;

      datosAdolfoFinal.push({ datos: datosAdolfo[i], formatos: formatosAdolfo[i] });
      nombresEnAdolfo.add(nombre.toString().trim().toUpperCase());
    }
    Logger.log(`RAW_ADOLFO: ${datosAdolfoFinal.length} filas, ${nombresEnAdolfo.size} nombres únicos`);
  }

  // PASO 2: Procesar RAW omitiendo nombres ya en RAW_ADOLFO (si existe)
  const datosRawFinal = [];
  if (sheetRaw) {
    const datosRaw = sheetRaw.getDataRange().getValues();
    const formatosRaw = sheetRaw.getDataRange().getNumberFormats();

    for (let i = 1; i < datosRaw.length; i++) {
      const nombre = datosRaw[i][1];
      if (!nombre || nombre.toString().toUpperCase() === 'SIN USUARIO') continue;
      if (nombresEnAdolfo.has(nombre.toString().trim().toUpperCase())) continue;
      datosRawFinal.push({ datos: datosRaw[i], formatos: formatosRaw[i] });
    }
    Logger.log(`RAW filtrado: ${datosRawFinal.length} filas`);
  }

  // PASO 3: Escribir en RAW_COMBINADO (si hay algo que escribir)
  const numColumnas = sheetCombinado.getLastColumn();
  let filaDestino = 2;

  if (datosAdolfoFinal.length > 0) {
    sheetCombinado.getRange(filaDestino, 1, datosAdolfoFinal.length, numColumnas)
      .setValues(datosAdolfoFinal.map(i => i.datos))
      .setNumberFormats(datosAdolfoFinal.map(i => i.formatos));
    filaDestino += datosAdolfoFinal.length;
  }

  if (datosRawFinal.length > 0) {
    sheetCombinado.getRange(filaDestino, 1, datosRawFinal.length, numColumnas)
      .setValues(datosRawFinal.map(i => i.datos))
      .setNumberFormats(datosRawFinal.map(i => i.formatos));
  }

  const totalFilas = datosAdolfoFinal.length + datosRawFinal.length;

  // Forzar columnas I y J como texto plano (solo si hay datos)
  if (totalFilas > 0) {
    sheetCombinado.getRange(2, 9, totalFilas, 1).setNumberFormat('@');
    sheetCombinado.getRange(2, 10, totalFilas, 1).setNumberFormat('@');
  }

  Logger.log(`RAW_COMBINADO: ${totalFilas} filas totales (excluyendo choferes que se agregan en PASO 0B)`);
  return totalFilas;
}

function limpiarDatosRAW() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheetCombinado = ss.getSheetByName('RAW_COMBINADO');

  if (!sheetCombinado) {
    crearRawCombinado();
    sheetCombinado = ss.getSheetByName('RAW_COMBINADO');
    if (!sheetCombinado) throw new Error('No se pudo crear RAW_COMBINADO');
  }

  const sheetTurnos = ss.getSheetByName(CONFIG.HOJAS.TURNOS_DEFAULT);
  const usuariosValidos = obtenerUsuariosValidos(sheetTurnos);
  const datosCombinado = sheetCombinado.getDataRange().getValues();

  const datosLimpios = [];
  const registrosPorEmpleadoFecha = {};
  let procesados = 0, rechazados = 0;

  for (let i = 1; i < datosCombinado.length; i++) {
    const resultado = procesarFilaRAW(datosCombinado[i], usuariosValidos, registrosPorEmpleadoFecha, true);
    if (resultado) { datosLimpios.push(resultado.datos); procesados++; }
    else rechazados++;
  }

  Logger.log(`Procesados: ${procesados}, Rechazados: ${rechazados}`);
  return { datos: datosLimpios, registrosPorEmpleadoFecha };
}

function procesarFilaRAW(fila, usuariosValidos, registrosPorEmpleadoFecha, modoFlexible = false) {
  const idUsuario = fila[0];
  const nombreUsuario = fila[1];
  const fecha = fila[2];
  const turno = fila[3];
  const entrada = fila[8];
  const salida = fila[9];

  if (!idUsuario || idUsuario === '' || idUsuario === 0) return null;
  if (!nombreUsuario || nombreUsuario === '') return null;
  if (nombreUsuario.toString().toUpperCase() === 'SIN USUARIO') return null;
  if (turno === 'SIN HORARIO' || !turno) return null;
  if (!modoFlexible && !usuariosValidos[idUsuario]) return null;
  if (!fecha) return null;

  const fechaObj = new Date(fecha);
  const fechaStr = formatearFecha(fechaObj);
  const clave = `${idUsuario}_${fechaStr}`;
  registrosPorEmpleadoFecha[clave] = true;

  const tieneEntrada = entrada && entrada !== '';
  const tieneSalida = salida && salida !== '';

  if (!tieneEntrada && !tieneSalida) {
    const diaSemana = fechaObj.getDay();
    return {
      datos: {
        idUsuario, nombreUsuario, fecha: fechaObj, turno, checadas: [], numChecadas: 0,
        esFaltaAutomatica: diaSemana !== 0 && diaSemana !== 6,
        esDiaInhabil: diaSemana === 0 || diaSemana === 6
      }
    };
  }

  const todasLasChecadas = [...extraerHoras(entrada), ...extraerHoras(salida)]
    .filter(Boolean)
    .sort((a, b) => convertirHoraAMinutos(a) - convertirHoraAMinutos(b));

  const checadasFiltradas = filtrarChecadasDuplicadas(todasLasChecadas);

  if (checadasFiltradas.length === 0) {
    return { datos: { idUsuario, nombreUsuario, fecha: fechaObj, turno, checadas: [], numChecadas: 0, esFaltaAutomatica: true } };
  }

  return { datos: { idUsuario, nombreUsuario, fecha: fechaObj, turno, checadas: checadasFiltradas, numChecadas: checadasFiltradas.length } };
}

function obtenerUsuariosValidos(sheetTurnos) {
  if (!sheetTurnos) return {};
  const datos = sheetTurnos.getDataRange().getValues();
  const usuarios = {};
  for (let i = 1; i < datos.length; i++) {
    const [id, nombre, turno, , idAdolfo] = datos[i];
    if (id && nombre && turno && turno !== 'SIN HORARIO') {
      usuarios[id] = { nombre, turno };
      if (idAdolfo && idAdolfo !== id) {
        usuarios[idAdolfo] = { nombre, turno, esIdAdolfo: true, idReal: id };
      }
    }
  }
  return usuarios;
}

function extraerHoras(celda) {
  if (!celda || celda === '') return [];
  const texto = celda.toString().trim();
  const horas = [];
  const patron = /(\d{1,2}):(\d{2})(?::(\d{2}))?/g;
  let match;
  while ((match = patron.exec(texto)) !== null) {
    const h = parseInt(match[1]), m = parseInt(match[2]);
    if (h >= 0 && h <= 23 && m >= 0 && m <= 59) {
      horas.push(`${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`);
    }
  }
  return horas;
}

function filtrarChecadasDuplicadas(checadas) {
  if (checadas.length <= 1) return checadas;
  const filtradas = [checadas[0]];
  for (let i = 1; i < checadas.length; i++) {
    if (Math.abs(convertirHoraAMinutos(checadas[i]) - convertirHoraAMinutos(filtradas[filtradas.length - 1])) >= CONFIG.REGLAS.MIN_TIEMPO_ENTRE_CHECADAS) {
      filtradas.push(checadas[i]);
    }
  }
  return filtradas;
}

function clasificarTodasLasChecadas(datosAgrupados) {
  const sheetConfig = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.HOJAS.CONFIG_TURNOS);
  const configuracionTurnos = cargarConfiguracionTurnos(sheetConfig);
  const resultado = [];

  for (const clave in datosAgrupados) {
    const registro = datosAgrupados[clave];
    const turnoConfig = obtenerConfiguracionTurno(registro.turno, registro.fecha, configuracionTurnos);

    if (!turnoConfig) {
      Logger.log(`ADVERTENCIA: Sin configuración para turno ${registro.turno}`);
      continue;
    }

    resultado.push({
      idUsuario: registro.idUsuario,
      nombreUsuario: registro.nombreUsuario,
      fecha: registro.fecha,
      turno: registro.turno,
      checadas: registro.checadas,
      clasificacion: clasificarChecadas(registro.checadas, turnoConfig, registro.fecha),
      configuracionTurno: turnoConfig
    });
  }
  return resultado;
}

function cargarConfiguracionTurnos(sheetConfig) {
  if (!sheetConfig) return {};
  const datos = sheetConfig.getDataRange().getValues();
  const config = {};
  for (let i = 1; i < datos.length; i++) {
    const turno = datos[i][0];
    if (!turno || turno === 'SIN HORARIO') continue;
    config[turno] = {
      turno, dias: datos[i][1],
      entradaMin: datos[i][2], entradaMax: datos[i][3], tolerancia: datos[i][4],
      desayunoMin: datos[i][5], desayunoMax: datos[i][6],
      desayunoRegresoMin: datos[i][7], desayunoRegresoMax: datos[i][8],
      comidaMin: datos[i][9], comidaMax: datos[i][10],
      comidaRegresoMin: datos[i][11], comidaRegresoMax: datos[i][12],
      salidaMin: datos[i][13]
    };
  }
  return config;
}

function obtenerConfiguracionTurno(turno, fecha, configuracionTurnos) {
  const turnos = turno.split(',').map(t => t.trim());
  const diaSemana = obtenerDiaSemana(fecha);
  for (const t of turnos) {
    const config = configuracionTurnos[t];
    if (config && verificarDiaLaboral(config.dias, diaSemana)) return config;
  }
  return configuracionTurnos[turnos[0]];
}

function verificarDiaLaboral(diasConfig, dia) {
  if (!diasConfig) return false;
  return diasConfig.toLowerCase().includes(dia.toLowerCase());
}

function verificarDiaLaboralHelper(diasConfig, diaSemana) {
  const diasSemana = ['D', 'L', 'M', 'X', 'J', 'V', 'S'];
  const diaActual = diasSemana[diaSemana];
  if (diasConfig.includes('-')) {
    const [diaInicio, diaFin] = diasConfig.split('-');
    return diaSemana >= diasSemana.indexOf(diaInicio.trim()) && diaSemana <= diasSemana.indexOf(diaFin.trim());
  }
  return diasConfig.includes(diaActual);
}

function clasificarChecadas(checadas, turnoConfig, fecha) {
  const clasificacion = {
    entrada: null, salidaDesayuno: null, regresoDesayuno: null,
    salidaComida: null, regresoComida: null, salida: null,
    checadasNoClasificadas: [], faltaSalidaFinal: false, salioTemprano: false
  };

  if (!checadas || checadas.length === 0) return clasificacion;

  const pendientes = [...checadas];
  const tieneDesayuno = turnoConfig.desayunoMin && turnoConfig.desayunoMax;
  const tieneComida = turnoConfig.comidaMin && turnoConfig.comidaMax;

  // 1. Entrada
  clasificacion.entrada = pendientes.shift();

  // 2. Intermedias (todo excepto la última)
  const intermedias = pendientes.slice(0, pendientes.length - 1);

  for (const checada of intermedias) {
    const minutos = convertirHoraAMinutos(checada);
    let clasificada = false;

    if (tieneDesayuno) {
      const desayunoMinMin = convertirHoraAMinutos(formatearHora(turnoConfig.desayunoMin));
      const desayunoMaxMin = convertirHoraAMinutos(formatearHora(turnoConfig.desayunoMax));

      if (!clasificacion.salidaDesayuno && minutos >= desayunoMinMin && minutos <= desayunoMaxMin) {
        clasificacion.salidaDesayuno = checada; clasificada = true; continue;
      }
      if (!clasificacion.regresoDesayuno && clasificacion.salidaDesayuno) {
        const dif = minutos - convertirHoraAMinutos(clasificacion.salidaDesayuno);
        const maxDes = (parseInt(turnoConfig.desayunoRegresoMax) || 30) * 2;
        if (dif > 0 && dif <= maxDes) { clasificacion.regresoDesayuno = checada; clasificada = true; continue; }
      }
    }

    if (tieneComida && !clasificada) {
      const comidaMinMin = convertirHoraAMinutos(formatearHora(turnoConfig.comidaMin));
      const comidaMaxMin = convertirHoraAMinutos(formatearHora(turnoConfig.comidaMax));

      if (!clasificacion.salidaComida && minutos >= comidaMinMin && minutos <= comidaMaxMin) {
        clasificacion.salidaComida = checada; clasificada = true; continue;
      }
      if (!clasificacion.regresoComida && clasificacion.salidaComida) {
        const dif = minutos - convertirHoraAMinutos(clasificacion.salidaComida);
        const maxCom = (parseInt(turnoConfig.comidaRegresoMax) || 80) * 2;
        if (dif > 0 && dif <= maxCom) { clasificacion.regresoComida = checada; clasificada = true; continue; }
      }
    }

    if (!clasificada) clasificacion.checadasNoClasificadas.push(checada);
  }

  // 3. Salida final (última checada)
  if (pendientes.length > 0) {
    const ultima = pendientes[pendientes.length - 1];
    const minutosUltima = convertirHoraAMinutos(ultima);
    const salidaOficial = convertirHoraAMinutos(formatearHora(turnoConfig.salidaMin));
    const limiteInferior = salidaOficial - CONFIG.REGLAS.VENTANA_SALIDA_MINUTOS;

    const esSalidaComida = ultima === clasificacion.salidaComida;
    const esSalidaDesayuno = ultima === clasificacion.salidaDesayuno;

    if (minutosUltima >= limiteInferior && !esSalidaComida && !esSalidaDesayuno) {
      clasificacion.salida = ultima;
      if (minutosUltima < salidaOficial - 30) {
        clasificacion.salioTemprano = true;
      }
    } else {
      clasificacion.faltaSalidaFinal = true;
      if (!esSalidaComida && !esSalidaDesayuno) {
        clasificacion.checadasNoClasificadas.push(ultima);
      }
    }
  } else if (clasificacion.entrada) {
    clasificacion.faltaSalidaFinal = true;
  }

  return clasificacion;
}

function cargarEventualidades() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.HOJAS.EVENTUALIDADES);

  const eventualidades = {
    vacaciones: [], enfermedades: [], diasFestivos: [],
    excepcionesRetardo: { turnos: [], ids: [] }
  };

  if (!sheet || sheet.getLastRow() <= 4) {
    Logger.log('EVENTUALIDADES: sin datos o no existe');
    return eventualidades;
  }

  const numFilas = sheet.getLastRow() - 4;
  const datos = sheet.getRange(5, 1, numFilas, 17).getValues();

  for (let i = 0; i < datos.length; i++) {
    // Vacaciones (cols A-D → índices 0-3)
    if (datos[i][0] && datos[i][2] && datos[i][3]) {
      eventualidades.vacaciones.push({
        idUsuario: datos[i][0], nombre: datos[i][1],
        fechaInicio: new Date(datos[i][2]), fechaFin: new Date(datos[i][3])
      });
    }

    // Enfermedades (cols F-K → índices 5-10)
    if (datos[i][5] && datos[i][7]) {
      eventualidades.enfermedades.push({
        idUsuario: datos[i][5], nombre: datos[i][6],
        fecha: new Date(datos[i][7]), tipo: datos[i][8] || 'DIA_COMPLETO',
        salioTemprano: datos[i][9] === 'SÍ' || datos[i][9] === 'SI',
        horaSalida: datos[i][10]
      });
    }

    // Días festivos (cols M-N → índices 12-13)
    if (datos[i][12]) {
      eventualidades.diasFestivos.push({
        fecha: new Date(datos[i][12]), descripcion: datos[i][13] || 'Día festivo'
      });
    }

    // Excepciones retardo (cols P-Q → índices 15-16)
    if (datos[i][15] && datos[i][16]) {
      const tipo = datos[i][15].toString().toUpperCase();
      const valor = datos[i][16];
      if (tipo === 'TURNO' && !eventualidades.excepcionesRetardo.turnos.includes(valor)) {
        eventualidades.excepcionesRetardo.turnos.push(valor);
      } else if (tipo === 'ID') {
        const id = parseInt(valor);
        if (!isNaN(id) && !eventualidades.excepcionesRetardo.ids.includes(id)) {
          eventualidades.excepcionesRetardo.ids.push(id);
        }
      }
    }
  }

  Logger.log(`Eventualidades: ${eventualidades.vacaciones.length} vac, ${eventualidades.enfermedades.length} enf, ${eventualidades.diasFestivos.length} festivos`);
  return eventualidades;
}

function verificarEventualidad(idUsuario, fecha, eventualidades) {
  const fechaStr = formatearFecha(fecha);

  for (const festivo of eventualidades.diasFestivos) {
    if (fechaStr === formatearFecha(festivo.fecha)) return { tipo: 'DIA_FESTIVO', data: festivo };
  }
  for (const vac of eventualidades.vacaciones) {
    if (vac.idUsuario == idUsuario && fechaStr >= formatearFecha(vac.fechaInicio) && fechaStr <= formatearFecha(vac.fechaFin)) {
      return { tipo: 'VACACIONES', data: vac };
    }
  }
  for (const enf of eventualidades.enfermedades) {
    if (enf.idUsuario == idUsuario && fechaStr === formatearFecha(enf.fecha)) {
      return { tipo: 'ENFERMEDAD', data: enf };
    }
  }
  return null;
}

function verificarExcepcionRetardo(idUsuario, turno, excepcionesRetardo) {
  for (const t of excepcionesRetardo.turnos) {
    if (turno && turno.includes(t)) return true;
  }
  return excepcionesRetardo.ids.includes(parseInt(idUsuario));
}

function calcularMetricasDiarias(checadasClasificadas) {
  const eventualidades = cargarEventualidades();
  return checadasClasificadas.map(registro => calcularMetricaRegistro(registro, eventualidades));
}

// ============================================================================
// Si un empleado activo de TURNOS_DEFAULT no aparece en RAW ni en RAW_ADOLFO
// para un día determinado del rango procesado, NO se le genera fila en
// METRICAS_DIARIAS. Esto rompe los reportes de faltas (renderFaltas, renderAlertas,
// WhatsApp) que filtran por `Es Falta = SÍ`, porque la fila simplemente no existe.
//
// Esta función completa el dataset: para cada (empleado, fecha) del rango que
// no tenga métrica, genera una métrica con `esFalta = true`, salvo que sea día
// inhábil para su turno (sábado/domingo según CONFIG_TURNOS), vacaciones o
// enfermedad registrada en EVENTUALIDADES.
// ============================================================================
function completarEmpleadosAusentes(metricas) {
  if (!metricas || metricas.length === 0) return [];

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetTurnos = ss.getSheetByName('TURNOS_DEFAULT');
  const sheetConfigTurnos = ss.getSheetByName('CONFIG_TURNOS');
  if (!sheetTurnos || !sheetConfigTurnos) return [];

  // 1. Determinar el rango de fechas presente en las métricas existentes
  const fechasSet = new Set();
  let fechaMin = null, fechaMax = null;
  metricas.forEach(m => {
    if (!(m.fecha instanceof Date)) return;
    const key = Utilities.formatDate(m.fecha, TIMEZONE, 'yyyy-MM-dd');
    fechasSet.add(key);
    if (!fechaMin || m.fecha < fechaMin) fechaMin = new Date(m.fecha);
    if (!fechaMax || m.fecha > fechaMax) fechaMax = new Date(m.fecha);
  });
  if (!fechaMin || !fechaMax) return [];

  // 2. Construir set de (idUsuario_fechaStr) que YA están en métricas
  const clavesPresentes = new Set();
  metricas.forEach(m => {
    if (!(m.fecha instanceof Date) || !m.idUsuario) return;
    const fechaStr = Utilities.formatDate(m.fecha, TIMEZONE, 'yyyy-MM-dd');
    clavesPresentes.add(m.idUsuario + '_' + fechaStr);
  });

  // 3. Cargar configuración de turnos y eventualidades
  const configuracionTurnos = cargarConfiguracionTurnos(sheetConfigTurnos);
  const eventualidades = cargarEventualidades();

  // 4. Recorrer empleados activos de TURNOS_DEFAULT y rellenar huecos
  const datosTurnos = sheetTurnos.getDataRange().getValues();
  const nuevasMetricas = [];

  // Lista de fechas del rango (todas las que aparecen en al menos una métrica)
  const fechasArray = Array.from(fechasSet).sort();

  for (let i = 1; i < datosTurnos.length; i++) {
    const idUsuario = datosTurnos[i][0];
    const nombre    = (datosTurnos[i][1] || '').toString().trim();
    const turno     = (datosTurnos[i][2] || '').toString().trim();
    if (!idUsuario || !nombre || !turno || turno === 'SIN HORARIO') continue;

    for (const fechaStr of fechasArray) {
      const clave = idUsuario + '_' + fechaStr;
      if (clavesPresentes.has(clave)) continue; // ya tiene métrica

      // Construir Date desde fechaStr "yyyy-MM-dd"
      const partes = fechaStr.split('-').map(Number);
      const fecha = new Date(partes[0], partes[1] - 1, partes[2]);

      // Buscar configuración de turno aplicable a esta fecha
      const turnoConfig = obtenerConfiguracionTurno(turno, fecha, configuracionTurnos);
      if (!turnoConfig) continue;

      // Verificar si es día laboral para su turno; si no, marcar como INHÁBIL
      const diaSemana = obtenerDiaSemana(fecha);
      const esDiaLaboral = verificarDiaLaboral(turnoConfig.dias, diaSemana);

      // Verificar eventualidad (vacaciones, enfermedad, día festivo)
      const eventualidad = verificarEventualidad(idUsuario, fecha, eventualidades);

      const metrica = {
        idUsuario: idUsuario,
        nombreUsuario: nombre,
        fecha: fecha,
        turno: turno,
        checadasOriginales: [],
        entrada: null, salidaDesayuno: null, regresoDesayuno: null,
        salidaComida: null, regresoComida: null, salida: null,
        minutosRetardoEntrada: 0,
        estatusEntrada: 'FALTA',
        minutosDesayuno: 0, minutosComida: 0,
        excedioComida: false, excedioDesayuno: false,
        horasTrabajadas: 0, horasExtra: 0,
        esRetardo: false, esFalta: false, esVacaciones: false,
        esEnfermedad: false, esDiaInhabil: false,
        observaciones: []
      };

      // Aplicar reglas en orden: día inhábil > eventualidad > falta
      if (!esDiaLaboral) {
        metrica.esDiaInhabil = true;
        metrica.estatusEntrada = 'INHÁBIL';
        const dias = ['Domingo','Lunes','Martes','Miércoles','Jueves','Viernes','Sábado'];
        metrica.observaciones.push('📅 ' + dias[diaSemana]);
      } else if (eventualidad) {
        if (eventualidad.tipo === 'DIA_FESTIVO') {
          metrica.esDiaInhabil = true;
          metrica.estatusEntrada = 'DÍA FESTIVO';
          metrica.observaciones.push('🎉 Día festivo');
        } else if (eventualidad.tipo === 'VACACIONES') {
          metrica.esVacaciones = true;
          metrica.estatusEntrada = 'VACACIONES';
          metrica.observaciones.push('🏖️ Vacaciones');
        } else if (eventualidad.tipo === 'ENFERMEDAD') {
          metrica.esEnfermedad = true;
          metrica.estatusEntrada = 'ENFERMEDAD';
          metrica.observaciones.push('🏥 Enfermedad');
        } else {
          // Eventualidad desconocida → tratar como falta
          metrica.esFalta = true;
          metrica.observaciones.push('❌ Sin checada (eventualidad: ' + eventualidad.tipo + ')');
        }
      } else {
        // Día laboral SIN checada SIN eventualidad → FALTA
        metrica.esFalta = true;
        metrica.observaciones.push('❌ Sin checada — empleado no se presentó');

        // Calcular jornada esperada
        const minJornada = Math.max(0,
          convertirHoraAMinutos(formatearHora(turnoConfig.salidaMin)) -
          convertirHoraAMinutos(formatearHora(turnoConfig.entradaMin))
        );
        metrica.horasTrabajadas = (minJornada / 60).toFixed(2);
      }

      nuevasMetricas.push(metrica);
    }
  }

  return nuevasMetricas;
}

function calcularMetricaRegistro(registro, eventualidades) {
  const c = registro.clasificacion;
  const turnoConfig = registro.configuracionTurno;
  const diaSemana = registro.fecha.getDay();
  const esFinDeSemana = diaSemana === 0 || diaSemana === 6;
  const eventualidad = verificarEventualidad(registro.idUsuario, registro.fecha, eventualidades);

  const metrica = {
    idUsuario: registro.idUsuario, nombreUsuario: registro.nombreUsuario,
    fecha: registro.fecha, turno: registro.turno,
    checadasOriginales: registro.checadas,
    entrada: c.entrada, salidaDesayuno: c.salidaDesayuno, regresoDesayuno: c.regresoDesayuno,
    salidaComida: c.salidaComida, regresoComida: c.regresoComida, salida: c.salida,
    minutosRetardoEntrada: 0, estatusEntrada: 'FALTA',
    minutosDesayuno: 0, minutosComida: 0,
    excedioComida: false, excedioDesayuno: false,
    horasTrabajadas: 0, horasExtra: 0,
    esRetardo: false, esFalta: false, esVacaciones: false,
    esEnfermedad: false, esDiaInhabil: false,
    observaciones: []
  };

  if (esFinDeSemana || registro.esDiaInhabil) {
    metrica.esDiaInhabil = true;
    metrica.estatusEntrada = 'INHÁBIL';
    metrica.observaciones.push(diaSemana === 0 ? '📅 Domingo' : '📅 Sábado');
    return metrica;
  }

  if (eventualidad) {
    if (eventualidad.tipo === 'DIA_FESTIVO') {
      metrica.esDiaInhabil = true;
      metrica.estatusEntrada = 'DÍA FESTIVO';
      metrica.observaciones.push(`🎉 ${eventualidad.data.descripcion}`);
      return metrica;
    }
    if (eventualidad.tipo === 'VACACIONES') {
      metrica.esVacaciones = true;
      metrica.estatusEntrada = 'VACACIONES';
      metrica.observaciones.push('🏖️ Vacaciones');
      return metrica;
    }
    if (eventualidad.tipo === 'ENFERMEDAD') {
      metrica.esEnfermedad = true;
      if (eventualidad.data.tipo === 'DIA_COMPLETO') {
        metrica.estatusEntrada = 'ENFERMEDAD';
        metrica.observaciones.push('🏥 Enfermedad - Día completo');
        return metrica;
      }
      metrica.observaciones.push('🏥 Enfermedad - Salió temprano');
    }
  }

  const tieneExcepcion = verificarExcepcionRetardo(registro.idUsuario, registro.turno, eventualidades.excepcionesRetardo);

  if (c.entrada) {
    if (tieneExcepcion) {
      metrica.estatusEntrada = 'PUNTUAL';
      metrica.observaciones.push('⏰ Excepción de retardo (no se contabiliza)');
    } else {
      const resultado = calcularRetardoEntrada(c.entrada, turnoConfig);
      metrica.minutosRetardoEntrada = resultado.minutos;
      metrica.estatusEntrada = resultado.estatus;
      metrica.esRetardo = resultado.estatus === 'RETARDO';
    }
    metrica.esFalta = false;
  } else if (!metrica.esVacaciones && !metrica.esEnfermedad) {
    metrica.esFalta = true;
    const minJornada = Math.max(0,
      convertirHoraAMinutos(formatearHora(turnoConfig.salidaMin)) -
      convertirHoraAMinutos(formatearHora(turnoConfig.entradaMin))
    );
    metrica.horasTrabajadas = (minJornada / 60).toFixed(2);
    metrica.observaciones.push(`Sin checada de entrada - Jornada esperada: ${metrica.horasTrabajadas} hrs`);
  }

  // Desayuno
  if (c.salidaDesayuno && c.regresoDesayuno) {
    metrica.minutosDesayuno = calcularDiferenciaMinutos(c.salidaDesayuno, c.regresoDesayuno);
    const maxDes = parseInt(turnoConfig.desayunoRegresoMax) || 30;
    if (metrica.minutosDesayuno > maxDes) {
      metrica.excedioDesayuno = true;
      metrica.observaciones.push(`⏰ Excedió desayuno: ${metrica.minutosDesayuno - maxDes} min extra (máx: ${maxDes})`);
    }
  } else if (c.salidaDesayuno && !c.regresoDesayuno) {
    metrica.minutosDesayuno = null;
    metrica.excedioDesayuno = true;
    metrica.observaciones.push('⚠️ Salió a desayuno pero no registró regreso');
  }

  // Comida
  if (c.salidaComida && c.regresoComida) {
    metrica.minutosComida = calcularDiferenciaMinutos(c.salidaComida, c.regresoComida);
    const maxCom = parseInt(turnoConfig.comidaRegresoMax) || 75;
    if (metrica.minutosComida > maxCom) {
      metrica.excedioComida = true;
      metrica.observaciones.push(`⏰ Excedió comida: ${metrica.minutosComida - maxCom} min extra (máx: ${maxCom})`);
    }
  } else if (c.salidaComida && !c.regresoComida) {
    metrica.minutosComida = null;
    metrica.excedioComida = true;
    metrica.observaciones.push('⚠️ Salió a comida pero no registró regreso');
  }

  // Horas trabajadas
  let salidaCalculo = c.salida;
  if (c.faltaSalidaFinal) {
    salidaCalculo = formatearHora(turnoConfig.salidaMin);
    metrica.observaciones.push('⚠️ No checó salida - se usa hora oficial');
  } else if (c.salioTemprano) {
    metrica.observaciones.push('⚠️ Salió temprano');
  }

  if (c.entrada && salidaCalculo) {
    const minTotales = calcularDiferenciaMinutos(c.entrada, salidaCalculo);
    const minDescansos = (metrica.minutosDesayuno || 0) + (metrica.minutosComida || 0);
    metrica.horasTrabajadas = Math.max(0, (minTotales - minDescansos) / 60).toFixed(2);

    if (turnoConfig.salidaMin) {
      const minExtra = convertirHoraAMinutos(salidaCalculo) - convertirHoraAMinutos(formatearHora(turnoConfig.salidaMin));
      if (minExtra >= 30) {
        metrica.horasExtra = (minExtra / 60).toFixed(2);
        metrica.observaciones.push(`⭐ Horas extra: ${metrica.horasExtra} hrs`);
      }
    }
  } else if (metrica.esFalta) {
    const minJornada = Math.max(0,
      convertirHoraAMinutos(formatearHora(turnoConfig.salidaMin)) -
      convertirHoraAMinutos(formatearHora(turnoConfig.entradaMin))
    );
    metrica.horasTrabajadas = (minJornada / 60).toFixed(2);
    metrica.observaciones.push(
      `❌ Falta - Jornada esperada: ${metrica.horasTrabajadas} hrs (${formatearHora(turnoConfig.entradaMin)}-${formatearHora(turnoConfig.salidaMin)})`
    );
  }

  return metrica;
}

// Lógica oficial de tolerancia/retardo (medida desde entradaMin, ej. T1 = 7:00):
//   0–15 min  → PUNTUAL    (bono perfecto; min reportado = 0)
//   16–17 min → TOLERANCIA (tolerancia; min reportado = excedente sobre 15, ej. 7:16 → 1, 7:17 → 2)
//   18–30 min → SIN BONO   (sin bono pero aún no es retardo; min reportado = retraso completo)
//   31+ min   → RETARDO    (esRetardo = true; min reportado = excedente sobre 30, ej. 7:33 → 3)
//
// Razón del diseño:
//   - PUNTUAL: por definición no hay retraso reportable.
//   - TOLERANCIA y RETARDO: se reporta lo que pasa del corte del estado previo,
//     porque eso es lo que se grafica/compara en los reportes.
//   - SIN BONO: no se grafica ni se compara, se queda con el retraso bruto.
function calcularRetardoEntrada(horaEntrada, turnoConfig) {
  const minEntrada    = convertirHoraAMinutos(horaEntrada);
  const minEntradaMin = convertirHoraAMinutos(formatearHora(turnoConfig.entradaMin));

  // Minutos de retraso medidos desde la hora oficial de entrada
  const minRetraso = minEntrada - minEntradaMin;

  if (minRetraso <= 15)  return { minutos: 0,                estatus: 'PUNTUAL' };
  if (minRetraso <= 17)  return { minutos: minRetraso - 15,  estatus: 'TOLERANCIA' };
  if (minRetraso <= 30)  return { minutos: minRetraso,       estatus: 'SIN BONO' };
  return                        { minutos: minRetraso - 30,  estatus: 'RETARDO' };
}

function generarAlertas(metricas) {
  const alertas = [];
  const porEmpleado = {};

  for (const m of metricas) {
    if (!porEmpleado[m.idUsuario]) porEmpleado[m.idUsuario] = { nombreUsuario: m.nombreUsuario, registros: [] };
    porEmpleado[m.idUsuario].registros.push(m);
  }

  
  for (const idUsuario in porEmpleado) {
    const { nombreUsuario, registros } = porEmpleado[idUsuario];
    const retardosPorMes = {}, faltasPorMes = {}, faltasPorQuincena = {};

    for (const r of registros) {
      const mes = `${r.fecha.getFullYear()}-${String(r.fecha.getMonth() + 1).padStart(2, '0')}`;
      const quincena = r.fecha.getDate() <= 15 ? 'Q1' : 'Q2';
      const claveQ = `${mes}-${quincena}`;

      if (!retardosPorMes[mes]) retardosPorMes[mes] = 0;
      if (!faltasPorMes[mes]) faltasPorMes[mes] = 0;
      if (!faltasPorQuincena[claveQ]) faltasPorQuincena[claveQ] = 0;

      if (r.esRetardo) retardosPorMes[mes]++;
      if (r.esFalta) { faltasPorMes[mes]++; faltasPorQuincena[claveQ]++; }
    }

    for (const mes in retardosPorMes) {
  const total = retardosPorMes[mes];
  if (total >= CONFIG.REGLAS.RETARDOS_PARA_DESCUENTO) {
    const mediosCount = Math.floor(total / CONFIG.REGLAS.RETARDOS_PARA_DESCUENTO);
    const diasCompletos = Math.floor(mediosCount / 2);
    const tienesMedio = mediosCount % 2 !== 0;
    let descuento = '';
    if (diasCompletos > 0 && tienesMedio) descuento = `${diasCompletos} día${diasCompletos > 1 ? 's' : ''} y medio`;
    else if (diasCompletos > 0) descuento = `${diasCompletos} día${diasCompletos > 1 ? 's' : ''}`;
    else descuento = 'medio día';

console.log(`🔢 Retardos ${nombreUsuario} mes ${mes}: total=${total}, mediosCount=${mediosCount}, diasCompletos=${diasCompletos}, tienesMedio=${tienesMedio}, descuento=${descuento}`);

    alertas.push({
      tipo: 'RETARDOS', idUsuario, nombreUsuario, mes, cantidad: total,
      mensaje: `${total} retardos en el mes (Descuento: ${descuento})`,
      severidad: total >= 6 ? 'CRÍTICA' : 'ALTA'
    });
  }
}

    for (const mes in faltasPorMes) {
      if (faltasPorMes[mes] >= CONFIG.REGLAS.FALTAS_ALERTA) {
        alertas.push({
          tipo: 'FALTAS', idUsuario, nombreUsuario, mes, cantidad: faltasPorMes[mes],
          mensaje: `${faltasPorMes[mes]} faltas en el mes`, severidad: 'CRÍTICA'
        });
      }
    }

    for (const quincena in faltasPorQuincena) {
      if (faltasPorQuincena[quincena] >= 3) {
        const [anio, mes, q] = quincena.split('-');
        alertas.push({
          tipo: 'FALTAS_QUINCENA', idUsuario, nombreUsuario, mes: `${anio}-${mes}`,
          cantidad: faltasPorQuincena[quincena],
          mensaje: `${faltasPorQuincena[quincena]} faltas en ${q === 'Q1' ? 'Primera' : 'Segunda'} quincena de ${anio}-${mes}`,
          severidad: 'CRÍTICA'
        });
      }
    }

    
  }

  return alertas;
}

function generarResumenMensual(metricas) {
  const resumenPorEmpleado = {};

  for (const m of metricas) {
    if (!esDiaLaboral(m.fecha)) continue;

    const mes = `${m.fecha.getFullYear()}-${String(m.fecha.getMonth() + 1).padStart(2, '0')}`;
    const clave = `${m.idUsuario}_${mes}`;

    if (!resumenPorEmpleado[clave]) {
      resumenPorEmpleado[clave] = {
        idUsuario: m.idUsuario, nombreUsuario: m.nombreUsuario, mes,
        diasTrabajados: 0, diasPuntuales: 0, diasTolerancia: 0,
        diasRetardo: 0, diasFalta: 0, diasVacaciones: 0, diasEnfermedad: 0,
        totalHorasTrabajadas: 0, excesosComida: 0
      };
    }

    const r = resumenPorEmpleado[clave];
    if (!m.esFalta && !m.esVacaciones && !m.esDiaInhabil) r.diasTrabajados++;
    r.totalHorasTrabajadas += parseFloat(m.horasTrabajadas || 0);
    if (m.estatusEntrada === 'PUNTUAL') r.diasPuntuales++;
    if (m.estatusEntrada === 'TOLERANCIA') r.diasTolerancia++;
    if (m.estatusEntrada === 'RETARDO') r.diasRetardo++;
    if (m.esFalta && !m.esVacaciones) r.diasFalta++;
    if (m.esVacaciones) r.diasVacaciones++;
    if (m.excedioComida) r.excesosComida++;
  }

  // Agregar días de enfermedad desde EVENTUALIDADES
  for (const clave in resumenPorEmpleado) {
    const r = resumenPorEmpleado[clave];
    const [anio, mes] = r.mes.split('-').map(Number);
    const enf = obtenerDiasEnfermedad(r.nombreUsuario, new Date(anio, mes - 1, 1), new Date(anio, mes, 0));
    r.diasEnfermedad = enf.totalDias;
  }

  return Object.values(resumenPorEmpleado);
}

function obtenerDiasEnfermedad(nombreEmpleado, fechaInicio, fechaFin) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.HOJAS.EVENTUALIDADES);

  if (!sheet || sheet.getLastRow() <= 4) return { totalDias: 0, fechas: [] };

  const numFilas = sheet.getLastRow() - 4;
  const datos = sheet.getRange(5, 1, numFilas, 11).getValues();
  const empleadoNorm = nombreEmpleado.toString().trim().toLowerCase();
  const fechas = [];

  for (let i = 0; i < datos.length; i++) {
    const nombre = datos[i][6]; // Columna G (índice 6 del rango)
    const fecha = datos[i][7];  // Columna H (índice 7 del rango)
    if (!nombre || !fecha) continue;
    if (nombre.toString().trim().toLowerCase() !== empleadoNorm) continue;
    const f = new Date(fecha);
    if (f >= fechaInicio && f <= fechaFin) {
      fechas.push(f);
    }
  }

  return {
    totalDias: fechas.length,
    fechas: fechas.map(f => Utilities.formatDate(f, Session.getScriptTimeZone(), 'dd/MM/yyyy'))
  };
}

    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('PUNTUAL').setBackground(CONFIG.COLORES.PUNTUAL).setRanges([range]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('TOLERANCIA').setBackground(CONFIG.COLORES.TOLERANCIA).setRanges([range]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('SIN BONO').setBackground('#fff3cd').setRanges([range]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('RETARDO').setBackground(CONFIG.COLORES.RETARDO).setRanges([range]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('FALTA').setBackground(CONFIG.COLORES.FALTA).setRanges([range]).build()
  ]);
}

      return {

        fecha: new Date(fila[2]), turno: fila[3], checadas,
        clasificacion: {
          entrada: fila[4] || null, salidaDesayuno: fila[5] || null, regresoDesayuno: fila[6] || null,
          salidaComida: fila[7] || null, regresoComida: fila[8] || null, salida: fila[9] || null,
          checadasNoClasificadas: fila[11] ? fila[11].split(',').map(c => c.trim()).filter(Boolean) : []
        },
        configuracionTurno: obtenerConfiguracionTurno(fila[3], new Date(fila[2]), configTurnos),
        esFaltaAutomatica: false
      };
    });

    const metricas = calcularMetricasDiarias(registros);
    const alertas = generarAlertas(metricas);
    const resumen = generarResumenMensual(metricas);

    escribirMetricas(metricas);
    escribirAlertas(alertas);
    escribirResumen(resumen);

    const sheetDashboard = ss.getSheetByName('DASHBOARD');
    if (sheetDashboard) generarDashboard(sheetDashboard);

    ui.alert('✅ Recálculo Completado',
      `Métricas: ${metricas.length}\nAlertas: ${alertas.length}\n\nHojas METRICAS, ALERTAS, RESUMEN y DASHBOARD actualizadas.`,
      ui.ButtonSet.OK);

  } catch (error) {
    SpreadsheetApp.getUi().alert('❌ Error', error.message, ui.ButtonSet.OK);
    Logger.log('Error recalcularConEventualidades: ' + error.stack);
  }
}

function getSheetData(sheetName) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
    if (!sheet) return { error: true, message: `La hoja "${sheetName}" no existe`, headers: [], data: [], rowCount: 0 };
 
    const lastRow = sheet.getLastRow();
    const lastCol = sheet.getLastColumn();
 
    if (lastRow < 2 || lastCol < 1) return { error: false, message: 'Sin datos', headers: [], data: [], rowCount: 0 };
 
    const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    const data = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues().map(row =>
      row.map(cell => {
        if (cell === null || cell === undefined) return '';
        if (cell instanceof Date) {
          // ⭐ Si la fecha base es 1899-12-30 (hora sin fecha real), enviar solo HH:mm
          const año = cell.getFullYear();
          if (año <= 1900) {
            // Es una hora pura — formatear directamente con getHours/getMinutes
            const h = String(cell.getHours()).padStart(2, '0');
            const m = String(cell.getMinutes()).padStart(2, '0');
            return `${h}:${m}`;
          }
          // Es una fecha real — enviar como ISO string
          return cell.toISOString();
        }
        return cell === '' ? '' : cell;
      })
    );
 
    return { error: false, message: 'OK', headers, data, rowCount: data.length };
  } catch (error) {
    return { error: true, message: error.message, headers: [], data: [], rowCount: 0 };
  }
}

  const sheet = ss.getSheetByName('EVENTUALIDADES');
  if (!sheet) return { error: true, message: 'La hoja EVENTUALIDADES no existe' };

  const secciones = {
    'ENFERMEDADES': { colIndex: 6, numColumnas: 5, filaInicio: 13 },
    'DIAS_FESTIVOS': { colIndex: 12, numColumnas: 2, filaInicio: 13 }
  };

  const config = secciones[seccion];
  if (!config) return { error: true, message: 'Sección no válida: ' + seccion };

  const lastRow = sheet.getLastRow();
  let filaDestino = config.filaInicio;
  for (let i = config.filaInicio; i <= lastRow + 1; i++) {
    if (!sheet.getRange(i, config.colIndex + 1).getValue()) { filaDestino = i; break; }
  }

  if (data.length > 0) {
    sheet.getRange(filaDestino, config.colIndex + 1, data.length, config.numColumnas).setValues(data);
    return { error: false, rowsWritten: data.length, startRow: filaDestino };
  }

  return { error: true, message: 'No hay datos para escribir' };
}

    // si toda la quincena fue vacaciones/enfermedad/inhábil/sin métricas,
    // no aplica para bono (ni a favor ni en contra).
    if (emp.diasRealesQ1 > 0 && !emp.retardo18PlusQ1 && !emp.faltasQ1) {
      if (tolQ1.length === 0) {
        resultado.Q1.bono.push({ nombre, diasTrabajados: emp.diasRealesQ1 });
      } else {
        resultado.Q1.tolerancia.push({ nombre, detalles: tolQ1 });
      }
    }

    // Q2 — misma regla
    if (emp.diasRealesQ2 > 0 && !emp.retardo18PlusQ2 && !emp.faltasQ2) {
      if (tolQ2.length === 0) {
        resultado.Q2.bono.push({ nombre, diasTrabajados: emp.diasRealesQ2 });
      } else {
        resultado.Q2.tolerancia.push({ nombre, detalles: tolQ2 });
      }
    }
  });

  ['Q1','Q2'].forEach(q => {
    resultado[q].bono.sort((a,b) => a.nombre.localeCompare(b.nombre,'es'));
    resultado[q].tolerancia.sort((a,b) => a.nombre.localeCompare(b.nombre,'es'));
  });

  // DEBUG Eric
  Object.keys(empleados).forEach(function(nombre) {
    if (nombre.toLowerCase().includes('eric s')) {
      var emp = empleados[nombre];
      Logger.log('🔍 ERIC DEBUG: ' + nombre);
      Logger.log('   retardo18PlusQ1=' + emp.retardo18PlusQ1);
      Logger.log('   retardo18PlusQ2=' + emp.retardo18PlusQ2);
      Logger.log('   faltasQ1=' + emp.faltasQ1);
      Logger.log('   faltasQ2=' + emp.faltasQ2);
      Logger.log('   detalleTolerancia=' + JSON.stringify(emp.detalleTolerancia));
      Logger.log('   enVacaciones=' + empleadosEnVacaciones.has(nombre));
    }
  });

  Logger.log('Q1 — Bono: ' + resultado.Q1.bono.length + ', Tolerancia: ' + resultado.Q1.tolerancia.length);
  Logger.log('Q2 — Bono: ' + resultado.Q2.bono.length + ', Tolerancia: ' + resultado.Q2.tolerancia.length);

  return {
    error: false,
    message: 'Bono calculado correctamente',
    headers: ['Q1_BONO','Q1_TOLERANCIA','Q2_BONO','Q2_TOLERANCIA'],
    data: [],
    rowCount: 0,
    metadata: {
      totalBono: resultado.Q1.bono.length,
      totalTolerancia: resultado.Q1.tolerancia.length,
      resultado: resultado
    }
  };
}

/**
 * Obtiene incidencias de empleados para una fecha específica
 * Lee AMBAS secciones: A4:D (manuales) + H4:J (Forms)
 */
function getIncidenciasEmpleados(fecha) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('INCIDENCIAS');
    if (!sheet) return { error: false, total: 0, incidencias: [] };

    const lastRow = sheet.getLastRow();
    if (lastRow < 4) return { error: false, total: 0, incidencias: [] };

    const incidencias = [];
    const traerTodas = (fecha === 'todas');

    let fechaStr = null;
    if (!traerTodas) {
      if (!fecha || fecha === 'undefined') fecha = getFechaActualMexico();
      if (typeof fecha === 'string' && fecha.match(/^\d{4}-\d{2}-\d{2}$/)) {
        fechaStr = fecha;
      } else {
        fechaStr = Utilities.formatDate(new Date(fecha), TIMEZONE, 'yyyy-MM-dd');
      }
    }

    // ============================================================================
    // SECCIÓN 1: INCIDENCIAS MANUALES (A4:D)
    // Columnas: A=Empleado, B=Actividad, C=Fecha y Hora Registro, D=Hora Final
    // ============================================================================
    const dataManuales = sheet.getRange(4, 1, lastRow - 3, 4).getValues();

    for (const row of dataManuales) {
      const empleado = row[0];
      const actividad = row[1];
      const fechaHora = row[2];

      if (!empleado || empleado.toString().trim() === '') continue;
      if (!fechaHora) continue;

      const fechaRegistroStr = parsearFechaIncidencia(fechaHora);
      if (!fechaRegistroStr) continue;

      if (!traerTodas && fechaRegistroStr !== fechaStr) continue;

      incidencias.push({
        empleado: empleado.toString().trim(),
        actividad: actividad ? actividad.toString().trim() : '',
        fecha: fechaRegistroStr,
        fuente: 'Manual'
      });
    }

    // ============================================================================
    // SECCIÓN 2: INCIDENCIAS FORMS (H4:J)
    // Columnas: H=Empleado, I=Actividad, J=Fecha y Hora de Registro
    // ============================================================================
    const dataForms = sheet.getRange(4, 8, lastRow - 3, 3).getValues(); // Columna H=8

    for (const row of dataForms) {
      const empleado = row[0]; // Columna H
      const actividad = row[1]; // Columna I
      const fechaHora = row[2]; // Columna J

      if (!empleado || empleado.toString().trim() === '') continue;
      if (!fechaHora) continue;

      const fechaRegistroStr = parsearFechaIncidencia(fechaHora);
      if (!fechaRegistroStr) continue;

      if (!traerTodas && fechaRegistroStr !== fechaStr) continue;

      incidencias.push({
        empleado: empleado.toString().trim(),
        actividad: actividad ? actividad.toString().trim() : '',
        fecha: fechaRegistroStr,
        fuente: 'Forms'
      });
    }

    Logger.log('📋 Incidencias encontradas: ' + incidencias.length);
    Logger.log('   - Manuales: ' + incidencias.filter(i => i.fuente === 'Manual').length);
    Logger.log('   - Forms: ' + incidencias.filter(i => i.fuente === 'Forms').length);

    return { error: false, total: incidencias.length, incidencias: incidencias };

  } catch (error) {
    Logger.log('ERROR getIncidenciasEmpleados: ' + error.toString());
    return { error: true, message: error.toString(), total: 0, incidencias: [] };
  }
}

/**
 * Parsea diferentes formatos de fecha a yyyy-MM-dd
 */
function parsearFechaIncidencia(fechaHora) {
  try {
    if (!fechaHora) return null;

    if (fechaHora instanceof Date) {
      return Utilities.formatDate(fechaHora, TIMEZONE, 'yyyy-MM-dd');
    }

    const fechaStr = fechaHora.toString();

    // Formato: "25/3/2026 14:16:38" o "25/03/2026"
    if (fechaStr.includes('/')) {
      const parteFecha = fechaStr.split(' ')[0];
      const partes = parteFecha.split('/');
      if (partes.length === 3) {
        const d = partes[0].padStart(2, '0');
        const m = partes[1].padStart(2, '0');
        const y = partes[2].length === 2 ? '20' + partes[2] : partes[2];
        return `${y}-${m}-${d}`;
      }
    }

    // Formato: "2026-03-25" o "2026-03-25T14:16:38"
    if (fechaStr.includes('-')) {
      return fechaStr.substring(0, 10);
    }

    return null;
  } catch (e) {
    return null;
  }
}

function generarTopHorasExtra(sheet, filaInicio) {
  _crearTituloTop(sheet, filaInicio, '⭐ TOP 5 - MÁS HORAS EXTRA', '#4caf50');
  filaInicio++;
  const sheetM = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.HOJAS.METRICAS_DIARIAS);
  if (!sheetM || sheetM.getLastRow() <= 1) { _sinDatos(sheet, filaInicio, 5); return; }

  const datos = sheetM.getDataRange().getValues();
  const h = datos[0];
  const [iN, iHE] = [h.indexOf('Nombre'), h.indexOf('Horas Extra')];
  const porEmp = {};

  for (let i = 1; i < datos.length; i++) {
    const nombre = datos[i][iN], hExtra = parseFloat(datos[i][iHE]) || 0;
    if (hExtra > 0) {
      if (!porEmp[nombre]) porEmp[nombre] = { totalHoras: 0, dias: 0 };
      porEmp[nombre].totalHoras += hExtra;
      porEmp[nombre].dias++;
    }
  }

  const ranking = Object.entries(porEmp).map(([n, v]) => ({ nombre: n, ...v }))
    .sort((a, b) => b.totalHoras - a.totalHoras).slice(0, 5);

  sheet.getRange(filaInicio, 1, 1, 5).setValues([['Nombre', 'Total Horas Extra', 'Días con Extra', 'Promedio/Día', 'Indicador']])
    .setBackground('#c8e6c9').setFontWeight('bold').setHorizontalAlignment('center');
  filaInicio++;

  if (!ranking.length) { _sinDatos(sheet, filaInicio, 5); return; }
  for (const emp of ranking) {
    sheet.getRange(filaInicio, 1, 1, 5).setValues([[
      emp.nombre, `${emp.totalHoras.toFixed(2)} hrs`, emp.dias,
      `${(emp.totalHoras / emp.dias).toFixed(2)} hrs`,
      '⭐'.repeat(Math.min(5, Math.ceil(emp.totalHoras / 5)))
    ]]).setBackground('#e8f5e9');
    filaInicio++;
  }
}

function generarTopExcesosComida(sheet, filaInicio) {
  _crearTituloTop(sheet, filaInicio, '🍽️ TOP 5 - MÁS EXCESOS EN COMIDA', '#ff9800');
  filaInicio++;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetM = ss.getSheetByName(CONFIG.HOJAS.METRICAS_DIARIAS);
  if (!sheetM || sheetM.getLastRow() <= 1) { _sinDatos(sheet, filaInicio, 5); return; }

  const configTurnos = cargarConfiguracionTurnos(ss.getSheetByName('CONFIG_TURNOS'));
  const datos = sheetM.getDataRange().getValues();
  const h = datos[0];
  const [iN, iT, iEC, iMC] = [h.indexOf('Nombre'), h.indexOf('Turno'), h.indexOf('Excedió Comida'), h.indexOf('Min. Comida')];
  const porEmp = {};

  for (let i = 1; i < datos.length; i++) {
    if (datos[i][iEC] !== 'SÍ') continue;
    const min = parseInt(datos[i][iMC]) || 0;
    if (!min) continue;
    const exceso = min - (parseInt((configTurnos[datos[i][iT]] || {}).comidaRegresoMax) || 80);
    if (exceso > 0) {
      const n = datos[i][iN];
      if (!porEmp[n]) porEmp[n] = { dias: 0, totalExceso: 0 };
      porEmp[n].dias++; porEmp[n].totalExceso += exceso;
    }
  }

  const ranking = Object.entries(porEmp).map(([n, v]) => ({ nombre: n, ...v }))
    .sort((a, b) => b.dias - a.dias).slice(0, 5);

  sheet.getRange(filaInicio, 1, 1, 5).setValues([['Nombre', 'Días Excedió', 'Total Min Extra', 'Promedio/Día', 'Nivel']])
    .setBackground('#ffe0b2').setFontWeight('bold').setHorizontalAlignment('center');
  filaInicio++;

  if (!ranking.length) { _sinDatos(sheet, filaInicio, 5); return; }
  for (const emp of ranking) {
    const nivel = emp.dias >= 5 ? '🔴 Alto' : emp.dias >= 3 ? '🟡 Medio' : '🟢 Bajo';
    sheet.getRange(filaInicio, 1, 1, 5).setValues([[
      emp.nombre, emp.dias, `${emp.totalExceso} min`,
      `${Math.round(emp.totalExceso / emp.dias)} min extra`, nivel
    ]]).setBackground(emp.dias >= 5 ? '#ffccbc' : '#fff3e0');
    filaInicio++;
  }
}

    if (exceso > 0) {
      const n = datos[i][iN];
      if (!porEmp[n]) porEmp[n] = { dias: 0, totalExceso: 0 };
      porEmp[n].dias++; porEmp[n].totalExceso += exceso;
    }
  }

  const ranking = Object.entries(porEmp).map(([n, v]) => ({ nombre: n, ...v }))
    .sort((a, b) => b.dias - a.dias).slice(0, 5);

  sheet.getRange(filaInicio, 1, 1, 5).setValues([['Nombre', 'Días Excedió', 'Total Min Extra', 'Promedio/Día', 'Nivel']])
    .setBackground('#ffccbc').setFontWeight('bold').setHorizontalAlignment('center');
  filaInicio++;

  if (!ranking.length) { _sinDatos(sheet, filaInicio, 5); return; }
  for (const emp of ranking) {
    const nivel = emp.dias >= 5 ? '🔴 Alto' : emp.dias >= 3 ? '🟡 Medio' : '🟢 Bajo';
    sheet.getRange(filaInicio, 1, 1, 5).setValues([[
      emp.nombre, emp.dias, `${emp.totalExceso} min`,
      `${Math.round(emp.totalExceso / emp.dias)} min extra`, nivel
    ]]).setBackground(emp.dias >= 5 ? '#ffab91' : '#ffccbc');
    filaInicio++;
  }
}

function getEnfermedades() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetEvent = ss.getSheetByName('EVENTUALIDADES');
  const sheetMetricas = ss.getSheetByName('METRICAS_DIARIAS');

  if (!sheetEvent || !sheetMetricas) return { error: true, enfermedades: [] };

  // ⭐ Obtener rango de fechas procesadas desde METRICAS_DIARIAS
  const datosM = sheetMetricas.getDataRange().getValues();
  const idxFecha = datosM[0].indexOf('Fecha');
  const fechas = datosM.slice(1)
    .map(r => r[idxFecha])
    .filter(f => f instanceof Date && !isNaN(f))
    .sort((a, b) => a - b);

  if (fechas.length === 0) return { error: true, enfermedades: [] };

  const fechaMin = fechas[0];
  const fechaMax = fechas[fechas.length - 1];

  // ⭐ Leer enfermedades desde fila 5, columnas F-K (índices 5-10)
  const lastRow = sheetEvent.getLastRow();
  if (lastRow < 5) return { error: false, enfermedades: [], fechaMin: fechaMin.toISOString(), fechaMax: fechaMax.toISOString() };

  const datos = sheetEvent.getRange(5, 1, lastRow - 4, 11).getValues();
  const enfermedades = [];

  datos.forEach(row => {
    const nombre = row[6]; // Columna G
    const fecha = row[7];  // Columna H
    if (!nombre || !fecha) return;

    const f = fecha instanceof Date ? fecha : new Date(fecha);
    if (isNaN(f)) return;

    // ⭐ Solo incluir si está dentro del rango procesado
    if (f >= fechaMin && f <= fechaMax) {
      enfermedades.push({
        nombre: nombre.toString().trim(),
        fecha: f.toISOString(),
        tipo: row[8] || 'DIA_COMPLETO'
      });
    }
  });

  return {
    error: false,
    enfermedades,
    fechaMin: fechaMin.toISOString(),
    fechaMax: fechaMax.toISOString()
  };
}

function getArchivosEnCarpeta(folderId) {
  try {
    const folder = DriveApp.getFolderById(folderId);
    const files = folder.getFiles();
    const lista = [];
    while (files.hasNext()) {
      const file = files.next();
      lista.push({
        id:       file.getId(),
        nombre:   file.getName(),
        fecha:    file.getLastUpdated().toLocaleDateString('es-MX', {
                    day: '2-digit', month: 'short', year: 'numeric',
                    hour: '2-digit', minute: '2-digit'
                  }),
        tamaño:   Math.round(file.getSize() / 1024) + ' KB'
      });
    }
    // Ordenar más reciente primero
    lista.sort((a, b) => new Date(b.fecha) - new Date(a.fecha));
    return { ok: true, archivos: lista };
  } catch(e) {
    return { ok: false, error: e.message };
  }
}

function inyectarDesdeArchivoDrive(fileId, sheetDestino, limpiar) {
  try {
    const file = DriveApp.getFileById(fileId);
    const blob = file.getBlob();
    const nombre = file.getName().toLowerCase();

    let filas = [];

    if (nombre.endsWith('.csv')) {
      const texto = blob.getDataAsString('UTF-8');
      const lineas = texto.split('\n').slice(1);
      filas = lineas
        .map(l => l.split(',').map(c => c.trim().replace(/^"|"$/g, '')))
        .filter(r => r.some(c => c !== ''));

    } else if (nombre.endsWith('.xls') || nombre.endsWith('.xlsx')) {
      const tempFile = Drive.Files.create(
        { name: '_temp_checador', mimeType: 'application/vnd.google-apps.spreadsheet' },
        blob
      );
      const tempId = tempFile.id;
      const tempSS = SpreadsheetApp.openById(tempId);
      const tempSheet = tempSS.getSheets()[0];
      const datos = tempSheet.getDataRange().getValues();
      filas = datos.slice(1).filter(r => r.some(c => c !== '' && c !== null));
      DriveApp.getFileById(tempId).setTrashed(true);

    } else {
      return { ok: false, error: 'Formato no soportado: ' + nombre };
    }

    if (filas.length === 0) return { ok: false, error: 'Sin datos en el archivo' };

    // ⭐ VACIAR COLUMNAS B (índice 1) y D (índice 3)
    Logger.log('📊 Total filas antes de vaciar B y D: ' + filas.length);
    
    filas = filas.map(fila => {
      const nuevaFila = [...fila]; // Copiar la fila
      
      // Vaciar columna B (índice 1)
      if (nuevaFila.length > 1) {
        nuevaFila[1] = '';
      }
      
      // Vaciar columna D (índice 3)
      if (nuevaFila.length > 3) {
        nuevaFila[3] = '';
      }
      
      return nuevaFila;
    });
    
    Logger.log('✅ Columnas B y D vaciadas en ' + filas.length + ' filas');

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hoja = ss.getSheetByName(sheetDestino);
    if (!hoja) return { ok: false, error: 'Hoja no encontrada: ' + sheetDestino };

    if (limpiar) {
      // ⭐ Limpiar TODAS las filas desde la 2, incluyendo vacías fantasma
      const totalFilas = hoja.getMaxRows();
      if (totalFilas >= 2) {
        hoja.getRange(2, 1, totalFilas - 1, hoja.getLastColumn() || 10).clearContent();
      }
      // Escribir siempre desde fila 2
      hoja.getRange(2, 1, filas.length, filas[0].length).setValues(filas);

    } else {
      // ⭐ Buscar última fila realmente ocupada (ignora vacías)
      const col1 = hoja.getRange(2, 1, hoja.getMaxRows() - 1, 1).getValues();
      let ultimaOcupada = 1;
      for (let i = 0; i < col1.length; i++) {
        if (col1[i][0] !== '' && col1[i][0] !== null) ultimaOcupada = i + 2;
      }
      hoja.getRange(ultimaOcupada + 1, 1, filas.length, filas[0].length).setValues(filas);
    }

    Logger.log('📝 Escritura completa en ' + sheetDestino + ': ' + filas.length + ' filas');
    return { ok: true, registros: filas.length };
    
  } catch(e) {
    Logger.log('❌ Error en inyectarDesdeArchivoDrive: ' + e.message);
    return { ok: false, error: e.message };
  }
}

    }
    
    // ============================================================================
    // DATOS DE METRICAS_DIARIAS (Asistencia, Faltas, Vacaciones)
    // ============================================================================
    const sheetMetricas = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('METRICAS_DIARIAS');
    if (!sheetMetricas) {
      Logger.log('❌ Hoja METRICAS_DIARIAS no encontrada');
      return;
    }
    
    const datos = sheetMetricas.getDataRange().getValues();
    const headers = datos[0];
    
    const idxFecha = headers.indexOf('Fecha');
    const idxNombre = headers.indexOf('Nombre');
    
    let idxEntrada = headers.indexOf('▶️ Entrada');
    if (idxEntrada === -1) idxEntrada = headers.findIndex(h => h && h.toString().includes('Entrada'));
    
      
    let idxVacaciones = headers.indexOf('Es Vacaciones');
    if (idxVacaciones === -1) idxVacaciones = headers.findIndex(h => h && h.toString().toLowerCase().includes('vacacion'));
    
    let idxRetardo = headers.indexOf('Es Retardo');
    if (idxRetardo === -1) idxRetardo = headers.findIndex(h => h && h.toString().toLowerCase().includes('retardo'));
    
    Logger.log('📍 Índices: Fecha=' + idxFecha + ', Nombre=' + idxNombre + ', Entrada=' + idxEntrada + ', Vacaciones=' + idxVacaciones + ', Retardo=' + idxRetardo);
    
    // Filtrar datos de hoy
    const datosHoy = datos.slice(1).filter(fila => {
      if (!fila[idxFecha]) return false;
      const fechaFila = new Date(fila[idxFecha]);
      const fechaFilaStr = Utilities.formatDate(fechaFila, TIMEZONE, 'yyyy-MM-dd');
      return fechaFilaStr === hoy;
    });
    
    Logger.log('📊 Registros de hoy: ' + datosHoy.length);
    
    if (datosHoy.length === 0) {
      Logger.log('⚠️ No hay datos para hoy');
      enviarWhatsApp(`⚠️ *ALERTA*\n\nNo hay datos de asistencia para hoy ${hoy}`);
      return;
    }
    
    // Empleados esperados
    const empleadosEsperados = [...new Set(datosHoy.map(fila => fila[idxNombre]))].filter(Boolean);
    const totalEsperados = empleadosEsperados.length;
    
    // ⭐ Empleados de VACACIONES (desde METRICAS_DIARIAS)
    const empleadosVacacionesMetricas = datosHoy.filter(fila => {
      if (idxVacaciones === -1) return false;
      const vacaciones = fila[idxVacaciones];
      return vacaciones === true || vacaciones === 'TRUE' || vacaciones === 'Sí' || vacaciones === 1 || vacaciones === '1';
    });
    let nombresVacaciones = [...new Set(empleadosVacacionesMetricas.map(fila => fila[idxNombre]))].filter(Boolean);
    
    // ⭐ Empleados PRESENTES (tienen entrada o salida)
    const empleadosPresentes = datosHoy.filter(fila => {
  const entrada = fila[idxEntrada];
  return entrada && entrada !== '' && entrada !== null;
});
    const nombresPresentes = [...new Set(empleadosPresentes.map(fila => fila[idxNombre]))].filter(Boolean).sort();
    const totalPresentes = nombresPresentes.length;
    
    // ⭐ Empleados con RETARDO
    const empleadosRetardo = datosHoy.filter(fila => {
  if (idxRetardo === -1) return false;
  const retardo = (fila[idxRetardo] || '').toString().trim();
  return retardo === 'SÍ' || retardo === 'SI' || retardo === 'Sí' || retardo === 'TRUE' || retardo === '1';
});
    const totalRetardos = [...new Set(empleadosRetardo.map(fila => fila[idxNombre]))].filter(Boolean).length;
    
    // ============================================================================
    // DATOS DE EVENTUALIDADES (Vacaciones y Enfermos)
    // ============================================================================
    let empleadosEnfermos = [];
    
    const sheetEventualidades = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('EVENTUALIDADES');
    if (sheetEventualidades) {
      const datosEvent = sheetEventualidades.getDataRange().getValues();
      
      // Parsear fecha de hoy (normalizado sin hora)
      const [anio, mes, dia] = hoy.split('-').map(Number);
      const hoyDate = new Date(anio, mes - 1, dia, 0, 0, 0);
      
      Logger.log('📅 Buscando eventualidades para: ' + hoy);
      
      // ⭐ VACACIONES (Columnas A-D, datos desde fila 13)
      // Columna A = ID, B = Nombre, C = Fecha Inicio, D = Fecha Fin
      for (let i = 12; i < datosEvent.length; i++) {
        const fila = datosEvent[i];
        const nombre = fila[1]; // Columna B
        const fechaInicio = fila[2]; // Columna C
        const fechaFin = fila[3]; // Columna D
        
        if (!nombre || !fechaInicio) continue;
        
        let inicio, fin;
        
        // Parsear fecha inicio
        if (fechaInicio instanceof Date) {
          inicio = fechaInicio;
        } else if (typeof fechaInicio === 'string') {
          const partes = fechaInicio.split(/[\/\-]/);
          if (partes.length === 3) {
            if (partes[0].length === 4) {
              inicio = new Date(partes[0], partes[1] - 1, partes[2], 12, 0, 0);
            } else {
              inicio = new Date(partes[2], partes[1] - 1, partes[0], 12, 0, 0);
            }
          }
        }
        
        // Parsear fecha fin
        if (fechaFin instanceof Date) {
          fin = fechaFin;
        } else if (typeof fechaFin === 'string') {
          const partes = fechaFin.split(/[\/\-]/);
          if (partes.length === 3) {
            if (partes[0].length === 4) {
              fin = new Date(partes[0], partes[1] - 1, partes[2], 12, 0, 0);
            } else {
              fin = new Date(partes[2], partes[1] - 1, partes[0], 12, 0, 0);
            }
          }
        }
        
        if (!fin) fin = inicio;
        
        const inicioNorm = inicio ? new Date(inicio.getFullYear(), inicio.getMonth(), inicio.getDate()) : null;
        const finNorm = fin ? new Date(fin.getFullYear(), fin.getMonth(), fin.getDate()) : null;
        if (inicioNorm && finNorm && hoyDate >= inicioNorm && hoyDate <= finNorm) {
          if (!nombresVacaciones.includes(nombre)) {
            nombresVacaciones.push(nombre);
            Logger.log('🏖️ Vacaciones encontrada: ' + nombre);
          }
        }
      }
      
      // ⭐ ENFERMEDADES (Columnas F-K, datos desde fila 13)
      // Columna F = ID, G = Nombre, H = Fecha, I = Tipo
      for (let i = 12; i < datosEvent.length; i++) {
        const fila = datosEvent[i];
        const nombre = fila[6]; // Columna G (índice 6)
        const fechaEnfermedad = fila[7]; // Columna H (índice 7)
        const tipo = fila[8]; // Columna I (índice 8)
        
        if (!nombre || !fechaEnfermedad) continue;
        
        let fechaEnf;
        
        if (fechaEnfermedad instanceof Date) {
          fechaEnf = Utilities.formatDate(fechaEnfermedad, TIMEZONE, 'yyyy-MM-dd');
        } else if (typeof fechaEnfermedad === 'string') {
          // Intentar parsear diferentes formatos
          if (fechaEnfermedad.includes('-')) {
            fechaEnf = fechaEnfermedad.substring(0, 10);
          } else if (fechaEnfermedad.includes('/')) {
            const partes = fechaEnfermedad.split('/');
            if (partes.length === 3) {
              if (partes[0].length === 4) {
                fechaEnf = fechaEnfermedad;
              } else {
                fechaEnf = `${partes[2]}-${partes[1].padStart(2, '0')}-${partes[0].padStart(2, '0')}`;
              }
            }
          }
        }
        
        if (fechaEnf === hoy) {
          if (!empleadosEnfermos.includes(nombre)) {
            empleadosEnfermos.push(nombre);
            Logger.log('🏥 Enfermo encontrado: ' + nombre + ' (' + tipo + ')');
          }
        }
      }
    }
    
    // Ordenar listas
    nombresVacaciones = nombresVacaciones.sort();
    empleadosEnfermos = empleadosEnfermos.sort();
    
    // ⭐ Empleados con turno vespertino — excluir si aún no es su hora de entrada
const EXCLUIR_ANTES_DE_HORA = [
  { nombre: 'Jairo Kayam Nava Mejía', horaEntrada: 13 } // Turno 13:00-18:00
];

const horaActual = new Date().getHours(); // Hora en servidor (UTC-6 ya está en TIMEZONE)

const empleadosAusentes = empleadosEsperados.filter(nombre => {
  // Si tiene turno vespertino y aún no llega su hora, excluir
  const excepcion = EXCLUIR_ANTES_DE_HORA.find(e => e.nombre === nombre);
  if (excepcion && horaActual < excepcion.horaEntrada) return false;
  
  return !nombresPresentes.includes(nombre) && 
         !nombresVacaciones.includes(nombre) &&
         !empleadosEnfermos.includes(nombre);
}).sort();
    const totalAusentes = empleadosAusentes.length;
    
    Logger.log('✅ Presentes: ' + totalPresentes);
    Logger.log('⏰ Retardos: ' + totalRetardos);
    Logger.log('🏖️ Vacaciones: ' + nombresVacaciones.length);
    Logger.log('🏥 Enfermos: ' + empleadosEnfermos.length);
    Logger.log('❌ Ausentes: ' + totalAusentes);
    
// ============================================================================
    // CONSTRUIR MENSAJE WHATSAPP
    // ============================================================================
    const porcentajeAsistencia = ((totalPresentes / totalEsperados) * 100).toFixed(1);

    // ── Departamentos ──
    const sheetTurnos2 = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('TURNOS_DEFAULT');
    const deptoPorNombre = {};
    const deptoPorNombreNorm = {}; // índice normalizado para fallback

    // Helper: limpieza agresiva de nombres
    //   - elimina NBSP (\u00A0), zero-width space (\u200B), tabs, etc.
    //   - colapsa espacios múltiples
    //   - trim
    function limpiarNombre(s) {
      if (!s) return '';
      return s.toString()
        .replace(/[\u00A0\u200B\u200C\u200D\uFEFF]/g, ' ')  // espacios "invisibles" → espacio normal
        .replace(/\s+/g, ' ')                                // colapsar espacios múltiples
        .trim();
    }
    function normalizarParaMatch(s) {
      return limpiarNombre(s).toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
    }

    if (sheetTurnos2) {
      const dt = sheetTurnos2.getDataRange().getValues();
      const hT = dt[0];
      const iNom  = hT.indexOf('Empleado');
      const iDept = hT.indexOf('DEPARTAMENTO');
      for (let i = 1; i < dt.length; i++) {
        const n = limpiarNombre(dt[i][iNom]);
        const dRaw = dt[i][iDept];
        // dRaw puede ser ' ' (espacio), '\t', null — todo eso debe caer a SIN DEPTO
        const d = limpiarNombre(dRaw);
        if (n) {
          deptoPorNombre[n] = d || 'SIN DEPTO';
          deptoPorNombreNorm[normalizarParaMatch(n)] = d || 'SIN DEPTO';
        }
      }
    }

    // Función de lookup con fallback normalizado (también limpia el nombre buscado)
    const _sinDeptoDetectados = new Set();  // para log al final
    const getDepto = function(nombre) {
      const nLimpio = limpiarNombre(nombre);
      if (deptoPorNombre[nLimpio] && deptoPorNombre[nLimpio] !== 'SIN DEPTO') {
        return deptoPorNombre[nLimpio];
      }
      const nNorm = normalizarParaMatch(nombre);
      if (deptoPorNombreNorm[nNorm] && deptoPorNombreNorm[nNorm] !== 'SIN DEPTO') {
        return deptoPorNombreNorm[nNorm];
      }
      // Cayó en SIN DEPTO — guardar para logging
      _sinDeptoDetectados.add(nLimpio || nombre);
      return 'SIN DEPTO';
    };

    // ── Faltas acumuladas desde METRICAS_DIARIAS (solo Lun-Vie) ──
    // faltasPorNombre: { nombre: { q1: N, q2: N, total: N } }
    const faltasPorNombre = {};
    const sheetMetricasF = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('METRICAS_DIARIAS');
    if (sheetMetricasF && sheetMetricasF.getLastRow() > 1) {
      const drM    = sheetMetricasF.getDataRange().getValues();
      const hM     = drM[0];
      const iNomM  = hM.indexOf('Nombre');
      const iFaltM = hM.indexOf('Es Falta');
      const iFechM = hM.indexOf('Fecha');
      const iInhM  = hM.indexOf('Es Día Inhábil');
      const iVacM  = hM.indexOf('Es Vacaciones');
      const iEnfM  = hM.indexOf('Es Enfermedad');
      for (let i = 1; i < drM.length; i++) {
        const n       = (drM[i][iNomM] || '').toString().trim();
        const esFalta = (drM[i][iFaltM] || '').toString().trim();
        if (!n || esFalta !== 'SÍ') continue;
        if (iInhM !== -1 && drM[i][iInhM] === 'SÍ') continue;
        if (iVacM !== -1 && drM[i][iVacM] === 'SÍ') continue;
        if (iEnfM !== -1 && drM[i][iEnfM] === 'SÍ') continue;
        const fechaF   = drM[i][iFechM];
        const fechaObj = fechaF instanceof Date ? fechaF : new Date(fechaF);
        const diaSem   = fechaObj.getDay();
        if (diaSem === 0 || diaSem === 6) continue;
        const diaMes   = fechaObj.getDate();
        if (!faltasPorNombre[n]) faltasPorNombre[n] = { q1: 0, q2: 0, total: 0 };
        if (diaMes <= 15) faltasPorNombre[n].q1++;
        else              faltasPorNombre[n].q2++;
        faltasPorNombre[n].total++;
      }
    }

    // ── Días de vacaciones hábiles: inicio, fin y día actual ──
    const vacInfoPorNombre = {};
    const sheetEvent2 = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('EVENTUALIDADES');
    if (sheetEvent2 && sheetEvent2.getLastRow() >= 13) {
      const devt = sheetEvent2.getDataRange().getValues();
      const [anioH, mesH, diaH] = hoy.split('-').map(Number);
      const hoyDateV = new Date(anioH, mesH - 1, diaH);

      const contarDiasHabiles = (inicio, fin) => {
        let count = 0;
        const cur = new Date(inicio);
        while (cur <= fin) {
          const d = cur.getDay();
          if (d !== 0 && d !== 6) count++;
          cur.setDate(cur.getDate() + 1);
        }
        return count;
      };

      const parseF = v => {
        if (v instanceof Date) return new Date(v.getFullYear(), v.getMonth(), v.getDate());
        const p = v.toString().split(/[\/\-]/);
        if (p.length !== 3) return null;
        return p[0].length === 4
          ? new Date(+p[0], +p[1]-1, +p[2])
          : new Date(+p[2], +p[1]-1, +p[0]);
      };

      for (let i = 12; i < devt.length; i++) {
        const nombre = (devt[i][1] || '').toString().trim();
        const rawIni = devt[i][2];
        const rawFin = devt[i][3];
        if (!nombre || !rawIni) continue;

        const ini = parseF(rawIni);
        const fin = parseF(rawFin) || ini;
        if (!ini) continue;

        if (hoyDateV >= ini && hoyDateV <= fin) {
          const totalDias  = contarDiasHabiles(ini, fin);
          const diaActual  = contarDiasHabiles(ini, hoyDateV);
          vacInfoPorNombre[nombre] = { diaActual, totalDias };
        }
      }
    }

    let mensaje = '📊 *REPORTE DE ASISTENCIA*\n';
    mensaje += '📅 ' + hoy + '\n\n';
    mensaje += '👥 *Total:* ' + totalEsperados + '\n';
    mensaje += '✅ *Asistieron:* ' + totalPresentes + ' (' + porcentajeAsistencia + '%)\n';
    mensaje += '⏰ *Retardos:* ' + totalRetardos + '\n';
    mensaje += '❌ *Ausentes:* ' + totalAusentes + '\n';
    mensaje += '🏖️ *Vacaciones:* ' + nombresVacaciones.length + '\n';
    mensaje += '🏥 *Enfermos:* ' + empleadosEnfermos.length + '\n';

    // ── Porcentaje de asistencia por departamento ──
    const deptoStats = {};
    empleadosEsperados.forEach(nombre => {
      const depto = getDepto(nombre);
      if (!deptoStats[depto]) deptoStats[depto] = { total: 0, presentes: 0 };
      deptoStats[depto].total++;
      const estaPresente = nombresPresentes.includes(nombre);
      const estaVacaciones = nombresVacaciones.includes(nombre);
      const estaEnfermo = empleadosEnfermos.includes(nombre);
      // Presentes + vacaciones + enfermos = no ausentes injustificados
      if (estaPresente || estaVacaciones || estaEnfermo) deptoStats[depto].presentes++;
    });

    mensaje += '\n📊 *ASISTENCIA POR DEPARTAMENTO:*\n';
    Object.keys(deptoStats).sort().forEach(depto => {
      const s = deptoStats[depto];
      const pct = ((s.presentes / s.total) * 100).toFixed(0);
      const barra = pct >= 90 ? '🟢' : pct >= 70 ? '🟡' : '🔴';
      mensaje += barra + ' _' + depto + '_: ' + s.presentes + '/' + s.total + ' (' + pct + '%)\n';
    });

    const nombresRetardo = [...new Set(empleadosRetardo.map(fila => fila[idxNombre]))].filter(Boolean).sort();
    if (nombresRetardo.length > 0) {
      mensaje += '\n⏰ *RETARDOS (' + nombresRetardo.length + '):*\n';
      mensaje += renderGrupos(agruparPorDepto(nombresRetardo, deptoPorNombre));
    }

    // ── AUSENTES con faltas acumuladas ──
    if (empleadosAusentes.length > 0) {
      mensaje += '\n❌ *AUSENTES (' + empleadosAusentes.length + '):*\n';
      const gruposAus = agruparPorDeptoConInfo(empleadosAusentes, deptoPorNombre, faltasPorNombre, null);
      mensaje += renderGruposConInfo(gruposAus);
    }

    // ── VACACIONES con día actual ──
    if (nombresVacaciones.length > 0) {
      mensaje += '\n🏖️ *VACACIONES (' + nombresVacaciones.length + '):*\n';
      const gruposVac = agruparPorDeptoConInfo(nombresVacaciones, deptoPorNombre, null, vacInfoPorNombre);
      mensaje += renderGruposConInfo(gruposVac);
    }

    if (empleadosEnfermos.length > 0) {
      mensaje += '\n🏥 *ENFERMOS (' + empleadosEnfermos.length + '):*\n';
      mensaje += renderGrupos(agruparPorDepto(empleadosEnfermos, deptoPorNombre));
    }

    if (totalAusentes === 0 && nombresVacaciones.length === 0 && empleadosEnfermos.length === 0) {
      mensaje += '\n🎉 *¡ASISTENCIA PERFECTA!* 🎉\n';
      mensaje += '✅ Todos asistieron hoy';
    }

    Logger.log('📱 Enviando mensaje de WhatsApp...');
    Logger.log('Longitud del mensaje: ' + mensaje.length + ' caracteres');

    // Log de empleados que terminaron en SIN DEPTO — útil para que el ADMIN
    // identifique a quién agregar/corregir el departamento en TURNOS_DEFAULT
    if (_sinDeptoDetectados.size > 0) {
      Logger.log('⚠️ Empleados sin departamento detectados en el reporte de WhatsApp:');
      _sinDeptoDetectados.forEach(n => Logger.log('   • "' + n + '"'));
    }

    const resultado = enviarWhatsApp(mensaje);

    if (resultado.ok) {
      Logger.log('✅ Reporte enviado exitosamente');
    } else {
      Logger.log('❌ Error al enviar: ' + resultado.message);
    }

    Logger.log('='.repeat(60));

  } catch (error) {
    Logger.log('❌ Error en enviarReporteDiario: ' + error.message);
    Logger.log(error.stack);
  }
}

  const hoy = getFechaActualMexico(); // ej: "2026-03-30"
  const [anio, mes, dia] = hoy.split('-').map(Number);
  const hoyDate = new Date(anio, mes - 1, dia, 12, 0, 0);
  
  Logger.log('🗓️ Hoy: ' + hoy);
  Logger.log('🗓️ hoyDate: ' + hoyDate);

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('EVENTUALIDADES');
  if (!sheet) { Logger.log('❌ Hoja no encontrada'); return; }

  const datos = sheet.getDataRange().getValues();

  Logger.log('--- VACACIONES (cols A-D desde fila 13) ---');
  for (let i = 12; i < datos.length; i++) {
    const fila = datos[i];
    const nombre = fila[1];
    const fechaInicio = fila[2];
    const fechaFin = fila[3];

    if (!nombre || !fechaInicio) continue;

    Logger.log(`Fila ${i+1} | Nombre: ${nombre} | Inicio: ${fechaInicio} (${typeof fechaInicio}) | Fin: ${fechaFin} (${typeof fechaFin})`);

    // Simular el parse que hace el código actual
    let inicio, fin;
    if (fechaInicio instanceof Date) {
      inicio = fechaInicio;
    } else if (typeof fechaInicio === 'string') {
      const partes = fechaInicio.split(/[\/\-]/);
      if (partes.length === 3) {
        inicio = partes[0].length === 4
          ? new Date(partes[0], partes[1]-1, partes[2], 12, 0, 0)
          : new Date(partes[2], partes[1]-1, partes[0], 12, 0, 0);
      }
    }
    if (fechaFin instanceof Date) {
      fin = fechaFin;
    } else if (typeof fechaFin === 'string') {
      const partes = fechaFin.split(/[\/\-]/);
      if (partes.length === 3) {
        fin = partes[0].length === 4
          ? new Date(partes[0], partes[1]-1, partes[2], 12, 0, 0)
          : new Date(partes[2], partes[1]-1, partes[0], 12, 0, 0);
      }
    }
    if (!fin) fin = inicio;

    const enRango = inicio && fin && hoyDate >= inicio && hoyDate <= fin;
    Logger.log(`   → inicio parseado: ${inicio} | fin parseado: ${fin} | ¿En rango?: ${enRango}`);
  }
}

function crearHojaConfigChoferes() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName('CONFIG_CHOFERES');
  if (sheet) return sheet;

  sheet = ss.insertSheet('CONFIG_CHOFERES');
  sheet.clear();

  // Título
  sheet.getRange('A1:G1').merge();
  sheet.getRange('A1')
    .setValue('⚙️ CONFIG CHOFERES — Zonas de Checada Válidas')
    .setBackground('#1a237e').setFontColor('#ffffff')
    .setFontSize(14).setFontWeight('bold')
    .setHorizontalAlignment('center').setVerticalAlignment('middle');
  sheet.setRowHeight(1, 40);

  // Instrucciones

// ============================================================================
// GUARDAR CHECADA CHOFER — Recibe foto en base64 + datos GPS
// ============================================================================

function guardarChecadaChofer(datos) {
  try {
    if (!datos) return { ok: false, message: 'No se recibieron datos' };

    crearHojaChecadorChoferes();
    crearHojaConfigChoferes();

    // ── HORA GENERADA EN EL SERVIDOR, NO EN EL CLIENTE ─────────────────────
    // Tablets / celulares pueden tener mal configurada la zona horaria
    // (ej. "America/Chicago" con DST en lugar de "America/Mexico_City"), lo que
    // antes causaba que la hora del cliente llegara +1h adelantada. Para
    // garantizar exactitud, GAS calcula la hora aquí con Utilities.formatDate
    // forzando el timezone de Ciudad de México. Esto se ejecuta en los
    // servidores de Google → la hora siempre es correcta, sin importar la
    // configuración del dispositivo que mande la checada.
    const ahora = new Date();
    const fechaServidor     = Utilities.formatDate(ahora, TIMEZONE, 'yyyy-MM-dd');
    const horaServidor      = Utilities.formatDate(ahora, TIMEZONE, 'HH:mm:ss');
    const timestampServidor = Utilities.formatDate(ahora, TIMEZONE, 'dd/MM/yyyy HH:mm:ss');

    // ── Escribir en CHECADOR_CHOFERES (8 columnas: A-H) ──
    const ss      = SpreadsheetApp.getActiveSpreadsheet();
    const sheet   = ss.getSheetByName('CHECADOR_CHOFERES');
    const lastRow = sheet.getLastRow();
    const fila    = lastRow + 1;

    sheet.getRange(fila, 1, 1, 8).setValues([[
      datos.idUsuario  || '',
      datos.nombre     || '',
      fechaServidor,
      horaServidor,
      timestampServidor,
      datos.lat        || '',
      datos.lng        || '',
      'VÁLIDA'  // ← Estado Zona SIEMPRE válida. Las coordenadas se siguen
                //   guardando y el cálculo de radio se usa solo para nombrar
                //   la zona, pero ninguna checada se rechaza por ubicación.
    ]]);

    // Formato TEXTO en Fecha (C), Hora (D) y Timestamp (E) para evitar shifts
    // de timezone al reinterpretar las cadenas como dates.
    sheet.getRange(fila, 3).setNumberFormat('@');
    sheet.getRange(fila, 4).setNumberFormat('@');
    sheet.getRange(fila, 5).setNumberFormat('@');

    Logger.log('✅ Checada guardada: ' + datos.nombre + ' a las ' + horaServidor +
               ' (cliente reportó: ' + (datos.hora || 'sin hora') + ')');
    return { ok: true, message: 'Checada registrada correctamente', horaServidor: horaServidor };

  } catch (e) {
    Logger.log('❌ Error guardarChecadaChofer: ' + e.message);
    return { ok: false, message: e.message };
  }
}

// ============================================================================
// FUSIONAR CHECADAS CHOFERES — Llamar después de crearRawCombinado()
// ============================================================================

function fusionarChecadasChoferes() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetCombinado = ss.getSheetByName('RAW_COMBINADO');
  const sheetChoferes  = ss.getSheetByName('CHECADOR_CHOFERES');

  if (!sheetCombinado) {
    Logger.log('❌ RAW_COMBINADO no existe');
    return 0;
  }
  if (!sheetChoferes || sheetChoferes.getLastRow() < 3) {
    Logger.log('ℹ️ CHECADOR_CHOFERES vacío, sin fusión');
    return 0;
  }

  const numColsChofer = sheetChoferes.getLastColumn();
const dataChoferes = sheetChoferes.getRange(3, 1, sheetChoferes.getLastRow() - 2, numColsChofer).getValues();

  const mapaChoferes = {};
  dataChoferes.forEach(row => {
    const nombre     = (row[1]  || '').toString().trim();
    const fecha      = row[2];
    const hora       = row[3];
    // Antes había filtro `if (estadoZona !== 'VÁLIDA') return;` pero ahora
    // TODAS las checadas son válidas (decisión de negocio: el radio solo
    // sirve para nombrar zonas, no para rechazar). Esto también recupera
    // checadas viejas que tenían "SIN_GPS" o estado en blanco.

    if (!nombre || !fecha || !hora) return;

    const fechaStr = fecha instanceof Date
      ? Utilities.formatDate(fecha, TIMEZONE, 'yyyy-MM-dd')
      : fecha.toString().substring(0, 10);

    const clave = nombre.toUpperCase() + '_' + fechaStr;
    if (!mapaChoferes[clave]) mapaChoferes[clave] = [];

    const horaStr = hora instanceof Date
      ? String(hora.getHours()).padStart(2,'0') + ':' + String(hora.getMinutes()).padStart(2,'0')
      : hora.toString().trim();

    mapaChoferes[clave].push(horaStr);
  });

  Logger.log('🚛 Claves válidas en CHECADOR_CHOFERES: ' + Object.keys(mapaChoferes).length);

  // Si CHECADOR_CHOFERES está vacío, no hay nada que fusionar ni agregar
  if (Object.keys(mapaChoferes).length === 0) {
    Logger.log('ℹ️ Sin claves válidas en CHECADOR_CHOFERES — fusión terminada');
    return 0;
  }

  const lastRowComb = sheetCombinado.getLastRow();
  // Si RAW_COMBINADO está vacío (caso: ya no hay RAW ni RAW_ADOLFO), saltar
  // directamente al segundo loop que agrega filas nuevas desde CHECADOR_CHOFERES.
  const tieneFilasComb = lastRowComb >= 2;
  const dataComb = tieneFilasComb
    ? sheetCombinado.getRange(2, 1, lastRowComb - 1, 10).getValues()
    : [];
  let fusionados = 0;

  // ── Helpers robustos ─────────────────────────────────────────────────────
  // Extrae TODAS las horas "HH:mm" de un string (acepta cualquier separador:
  // espacios, comas, tabs, "/", etc.) y normaliza el cero inicial.
  function extraerHoras(valor) {
    if (valor === null || valor === undefined) return [];
    // Si es Date, formatear directamente
    if (valor instanceof Date) {
      const h = valor.getHours(), m = valor.getMinutes();
      // Algunas celdas con time-only de GAS usan año 1899 — solo nos interesa hh:mm
      return [String(h).padStart(2, '0') + ':' + String(m).padStart(2, '0')];
    }
    const str = valor.toString();
    if (!str.trim()) return [];
    // Match global de patrones tipo H:MM o HH:MM (con segundos opcionales)
    const matches = str.match(/\b(\d{1,2}):(\d{2})(?::\d{2})?\b/g) || [];
    return matches.map(h => {
      const parts = h.split(':');
      const hh = Math.min(23, Math.max(0, parseInt(parts[0], 10) || 0));
      const mm = Math.min(59, Math.max(0, parseInt(parts[1], 10) || 0));
      return String(hh).padStart(2, '0') + ':' + String(mm).padStart(2, '0');
    });
  }

  // Convierte "HH:mm" a minutos totales — devuelve -1 si es inválido
  function hhmmToMin(s) {
    const m = (s || '').toString().match(/^(\d{1,2}):(\d{2})$/);
    if (!m) return -1;
    return parseInt(m[1], 10) * 60 + parseInt(m[2], 10);
  }

  // ── PRIMER LOOP: solo si hay filas en RAW_COMBINADO ───────────────────────
  if (tieneFilasComb) {
    // Acumulamos updates en memoria para escribir en UN solo setValues al final
    const updatesEntradaSalida = [];

    dataComb.forEach((row, idx) => {
      const nombre = (row[1] || '').toString().trim();
      const fecha  = row[2];
      if (!nombre || !fecha) return;

      const fechaStr = fecha instanceof Date
        ? Utilities.formatDate(fecha, TIMEZONE, 'yyyy-MM-dd')
        : fecha.toString().substring(0, 10);

      const clave = nombre.toUpperCase() + '_' + fechaStr;
      if (!mapaChoferes[clave]) return;

      const horasRaw = []
        .concat(extraerHoras(row[8]))   // Entrada
        .concat(extraerHoras(row[9]));  // Salida

      const horasChofer = (mapaChoferes[clave] || [])
        .map(h => extraerHoras(h))
        .reduce((acc, arr) => acc.concat(arr), []);

      const todasHoras = horasRaw.concat(horasChofer);
      if (todasHoras.length === 0) return;

      todasHoras.sort((a, b) => hhmmToMin(a) - hhmmToMin(b));

      const filtradas = [todasHoras[0]];
      for (let i = 1; i < todasHoras.length; i++) {
        const minPrev = hhmmToMin(filtradas[filtradas.length - 1]);
        const minCurr = hhmmToMin(todasHoras[i]);
        if (minPrev < 0 || minCurr < 0) continue;
        if (Math.abs(minCurr - minPrev) >= 5) filtradas.push(todasHoras[i]);
      }

      const nuevaEntrada = filtradas[0] || '';
      const nuevaSalida  = filtradas.slice(1).join(', ');

      updatesEntradaSalida.push({ fila: idx + 2, entrada: nuevaEntrada, salida: nuevaSalida });
      fusionados++;
    });

    // Aplicar todas las updates (escribir en cada fila individual — son pocas
    // típicamente, y no se pueden agrupar en un rango contiguo)
    updatesEntradaSalida.forEach(u => {
      sheetCombinado.getRange(u.fila, 9).setValue(u.entrada).setNumberFormat('@');
      sheetCombinado.getRange(u.fila, 10).setValue(u.salida).setNumberFormat('@');
    });
  }

  // ── Conjunto de claves YA presentes en RAW_COMBINADO (para no duplicar) ──
  const clavesEnCombinado = new Set(
    dataComb.map(row => {
      const nombre = (row[1] || '').toString().trim();
      const fecha  = row[2];
      if (!nombre || !fecha) return '';
      const fechaStr = fecha instanceof Date
        ? Utilities.formatDate(fecha, TIMEZONE, 'yyyy-MM-dd')
        : fecha.toString().substring(0, 10);
      return nombre.toUpperCase() + '_' + fechaStr;
    })
  );

  const sheetTurnos = ss.getSheetByName('TURNOS_DEFAULT');
  const turnoPorNombre = {};
  const idPorNombre    = {};
  // Helper de normalización: minúsculas sin acentos, espacios colapsados.
  // Hace matching tolerante a diferencias entre CHECADOR_CHOFERES y TURNOS_DEFAULT
  // (mayúsculas, acentos, NBSP, etc.).
  function _normMatch(s) {
    return (s || '').toString()
      .replace(/[\u00A0\u200B\u200C\u200D\uFEFF]/g, ' ')
      .replace(/\s+/g, ' ')
      .trim()
      .toLowerCase()
      .normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  }
  if (sheetTurnos) {
    const dt = sheetTurnos.getDataRange().getValues();
    const hT = dt[0];
    const iId     = hT.indexOf('Admin');
    const iNombre = hT.indexOf('Empleado');
    const iTurno  = hT.indexOf('Turno');
    for (let i = 1; i < dt.length; i++) {
      const n = (dt[i][iNombre] || '').toString().trim();
      if (n) {
        const key = _normMatch(n);
        turnoPorNombre[key] = (dt[i][iTurno] || '').toString().trim();
        idPorNombre[key]    = dt[i][iId];
      }
    }
  }

  const clavesAgregadas = new Set();
  let agregados = 0;

  // ── SEGUNDO LOOP: agregar filas nuevas para choferes que NO están en
  // RAW_COMBINADO. Acumulamos en memoria y escribimos en UN solo setValues
  // al final (era extremadamente lento hacerlo fila por fila — ~1seg c/u).
  const filasNuevas = [];
  Object.keys(mapaChoferes).forEach(clave => {
    if (clavesEnCombinado.has(clave) || clavesAgregadas.has(clave)) return;
    clavesAgregadas.add(clave);

    // La clave es "NOMBRE_FECHASTR" — extraemos ambos
    const sep = clave.lastIndexOf('_');
    const nombreUpper = clave.substring(0, sep);
    const fechaStr    = clave.substring(sep + 1);

    // Recuperar el nombre original (con su capitalización real) de dataChoferes
    const filaOriginal = dataChoferes.find(r =>
      (r[1] || '').toString().trim().toUpperCase() === nombreUpper
    );
    if (!filaOriginal) return;
    const nombre = (filaOriginal[1] || '').toString().trim();

    // Matching robusto contra TURNOS_DEFAULT
    const keyMatch = _normMatch(nombre);
    const turno = turnoPorNombre[keyMatch] || 'T3 Choferes';
    const id    = idPorNombre[keyMatch];
    if (!id) {
      Logger.log('⚠️ Chofer "' + nombre + '" no encontrado en TURNOS_DEFAULT — checada NO se fusiona');
      return;
    }

    const horas   = mapaChoferes[clave] || [];
    const entrada = horas[0] || '';
    const salida  = horas.slice(1).join(', ');
    const fechaDate = new Date(fechaStr + 'T12:00:00');

    filasNuevas.push([id, nombre, fechaDate, turno, '', '', '', '', entrada, salida]);
    agregados++;
  });

  // Escribir TODAS las filas nuevas en UN solo setValues (batch)
  if (filasNuevas.length > 0) {
    const filaInicio = sheetCombinado.getLastRow() + 1;
    sheetCombinado.getRange(filaInicio, 1, filasNuevas.length, 10).setValues(filasNuevas);
    // Forzar formato texto en columnas Entrada/Salida (col I y J)
    sheetCombinado.getRange(filaInicio, 9,  filasNuevas.length, 1).setNumberFormat('@');
    sheetCombinado.getRange(filaInicio, 10, filasNuevas.length, 1).setNumberFormat('@');
    Logger.log('➕ ' + filasNuevas.length + ' filas nuevas agregadas a RAW_COMBINADO desde CHECADOR_CHOFERES');
  }

  Logger.log('✅ Fusión: ' + fusionados + ' fusionados, ' + agregados + ' agregados');
  return fusionados + agregados;
}

function validarPin(pin) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    crearHojaUsuarios();
    const sheet   = ss.getSheetByName('USUARIOS');
    const lastRow = sheet.getLastRow();
    if (lastRow < 2) return { ok: false, message: 'Sin usuarios configurados' };

    const data = sheet.getRange(2, 1, lastRow - 1, 2).getValues();

    for (const row of data) {
      if (row[0].toString().trim() === pin.toString().trim()) {
        const idUsuario = row[1].toString().trim();
        if (idUsuario === 'ADMIN') return { ok: true, tipo: 'ADMIN' };

        const turnos = ss.getSheetByName('TURNOS_DEFAULT');
        if (!turnos) return { ok: false, message: 'TURNOS_DEFAULT no encontrada' };

        const dt = turnos.getDataRange().getValues();
        const hT = dt[0];
        const iId     = hT.indexOf('Admin');
        const iNombre = hT.indexOf('Empleado');

        for (let i = 1; i < dt.length; i++) {
          if (dt[i][iId].toString().trim() === idUsuario) {
            const nombre = dt[i][iNombre].toString().trim();
            // Verificar si ya tiene contraseña
            const tieneContrasena = verificarTieneContrasena(pin);
            return {
              ok: true,
              tipo: 'CHOFER',
              idUsuario,
              nombre,
              tieneContrasena
            };
          }
        }
        return { ok: false, message: 'ID no encontrado' };
      }
    }
    return { ok: false, message: 'PIN incorrecto' };
  } catch(e) {
    return { ok: false, message: e.message };
  }
}

function verificarTieneContrasena(pin) {
  try {
    crearHojaContrasenas();
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('CONTRASENAS_CHOFERES');
    const lastRow = sheet.getLastRow();
    if (lastRow < 3) return false;

    const data = sheet.getRange(3, 1, lastRow - 2, 1).getValues();
    return data.some(row => row[0].toString().trim() === pin.toString().trim());
  } catch(e) {
    return false;
  }
}

function validarContrasena(pin, contrasena) {
  try {
    crearHojaContrasenas();
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('CONTRASENAS_CHOFERES');
    const lastRow = sheet.getLastRow();
    if (lastRow < 3) return { ok: false, message: 'Sin contrasenas registradas' };

    const data = sheet.getRange(3, 1, lastRow - 2, 3).getValues();
    for (const row of data) {
      if (row[0].toString().trim() === pin.toString().trim()) {
        if (row[2].toString().trim() === contrasena.toString().trim()) {
          return { ok: true };
        } else {
          return { ok: false, message: 'Contraseña incorrecta' };
        }
      }
    }
    return { ok: false, message: 'PIN no encontrado en contraseñas' };
  } catch(e) {
    return { ok: false, message: e.message };
  }
}

function guardarContrasena(pin, nombre, contrasena) {
  try {
    crearHojaContrasenas();
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('CONTRASENAS_CHOFERES');

    // Verificar que no exista ya
    if (verificarTieneContrasena(pin)) {
      return { ok: false, message: 'Este PIN ya tiene contraseña registrada' };
    }

    const lastRow = sheet.getLastRow();
    const fila = lastRow + 1;
    // Forzar columnas A y C como texto ANTES de escribir
    sheet.getRange(fila, 1).setNumberFormat('@');
    sheet.getRange(fila, 3).setNumberFormat('@');
    sheet.getRange(fila, 1, 1, 3).setValues([[
      pin.toString().trim(),
      nombre.toString().trim(),
      contrasena.toString().trim()
    ]]);
    Logger.log('Contrasena guardada para: ' + nombre + ' PIN: ' + pin);
    return { ok: true };
  } catch(e) {
    return { ok: false, message: e.message };
  }
}

// ============================================================================
// getTodosLosUsuarios — Sync masivo para validación local en el cliente
// ============================================================================
// Devuelve TODOS los usuarios (PIN + ID + nombre + contraseña) en una sola
// llamada para que el frontend cachée en localStorage y valide instantáneamente
// sin volver a llamar a GAS. El ADMIN puede invocar sincronizarUsuariosManual()
// para forzar el resync cuando agregue/cambie usuarios.
// ============================================================================
function getTodosLosUsuarios() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    crearHojaUsuarios();
    crearHojaContrasenas();

    const sheetUsuarios     = ss.getSheetByName('USUARIOS');
    const sheetContrasenas  = ss.getSheetByName('CONTRASENAS_CHOFERES');
    const sheetTurnos       = ss.getSheetByName('TURNOS_DEFAULT');

    // ── 1. Construir mapa PIN → contraseña ─────────────────────────────────
    const mapaContrasenas = {};
    if (sheetContrasenas && sheetContrasenas.getLastRow() >= 3) {
      const dataC = sheetContrasenas.getRange(3, 1, sheetContrasenas.getLastRow() - 2, 3).getValues();
      dataC.forEach(row => {
        const pin = (row[0] || '').toString().trim();
        const cont = (row[2] || '').toString().trim();
        if (pin && cont) mapaContrasenas[pin] = cont;
      });
    }

    // ── 2. Construir mapa ID → nombre desde TURNOS_DEFAULT ────────────────
    const mapaTurnos = {};
    if (sheetTurnos) {
      const dt = sheetTurnos.getDataRange().getValues();
      const hT = dt[0];
      const iId     = hT.indexOf('Admin');
      const iNombre = hT.indexOf('Empleado');
      for (let i = 1; i < dt.length; i++) {
        const id     = (dt[i][iId]     || '').toString().trim();
        const nombre = (dt[i][iNombre] || '').toString().trim();
        if (id && nombre) mapaTurnos[id] = nombre;
      }
    }

    // ── 3. Leer USUARIOS y armar el listado final ─────────────────────────
    const usuarios = [];
    if (sheetUsuarios && sheetUsuarios.getLastRow() >= 2) {
      const dataU = sheetUsuarios.getRange(2, 1, sheetUsuarios.getLastRow() - 1, 2).getValues();
      dataU.forEach(row => {
        const pin = (row[0] || '').toString().trim();
        const idUsuario = (row[1] || '').toString().trim();
        if (!pin || !idUsuario) return;

        // ADMIN: tipo especial, sin contraseña adicional
        if (idUsuario === 'ADMIN') {
          usuarios.push({
            pin: pin,
            idUsuario: 'ADMIN',
            nombre: 'ADMIN',
            tipo: 'ADMIN',
            contrasena: mapaContrasenas[pin] || ''
          });
          return;
        }

        const nombre = mapaTurnos[idUsuario] || '';
        if (!nombre) return; // PIN sin empleado asociado en TURNOS_DEFAULT, se ignora

        usuarios.push({
          pin: pin,
          idUsuario: idUsuario,
          nombre: nombre,
          tipo: 'CHOFER',
          contrasena: mapaContrasenas[pin] || ''
        });
      });
    }

    Logger.log('✅ getTodosLosUsuarios: ' + usuarios.length + ' usuarios');
    return {
      ok: true,
      timestamp: new Date().toISOString(),
      total: usuarios.length,
      usuarios: usuarios
    };

  } catch(e) {
    Logger.log('❌ Error getTodosLosUsuarios: ' + e.message);
    return { ok: false, message: e.message, usuarios: [] };
  }
}

// ============================================================================
// VERSIONADO DE CACHE — Para que el frontend detecte cambios automáticamente
// ============================================================================
// Cuando el ADMIN modifica un PIN o contraseña en Sheets, el cache local de
// cada navegador queda desactualizado. Para evitar tener que pasar por cada
// dispositivo, usamos un timestamp guardado en PropertiesService:
//   - forzarResyncUsuarios()  → actualiza el timestamp (se llama desde el menú)
//   - getVersionUsuarios()    → devuelve el timestamp actual (lo llama el frontend)
// El frontend compara con el timestamp que tiene almacenado; si es más viejo,
// invalida el cache local y vuelve a llamar a getTodosLosUsuarios automáticamente.
// ============================================================================
function getVersionUsuarios() {
  try {
    const props = PropertiesService.getDocumentProperties();
    let version = props.getProperty('USUARIOS_CACHE_VERSION');
    // Si no existe (primera vez), sembrar con timestamp actual
    if (!version) {
      version = new Date().getTime().toString();
      props.setProperty('USUARIOS_CACHE_VERSION', version);
    }
    return { ok: true, version: version };
  } catch(e) {
    Logger.log('❌ Error getVersionUsuarios: ' + e.message);
    return { ok: false, message: e.message, version: '0' };
  }
}

function forzarResyncUsuarios() {
  try {
    const props = PropertiesService.getDocumentProperties();
    const nuevaVersion = new Date().getTime().toString();
    props.setProperty('USUARIOS_CACHE_VERSION', nuevaVersion);

    SpreadsheetApp.getUi().alert(
      '🔄 Usuarios sincronizados',
      'Los PINs y contraseñas se actualizarán automáticamente en todos los ' +
      'dispositivos la próxima vez que los empleados abran la app.\n\n' +
      'Versión nueva: ' + nuevaVersion,
      SpreadsheetApp.getUi().ButtonSet.OK
    );

    Logger.log('🔄 USUARIOS_CACHE_VERSION actualizada a ' + nuevaVersion);
    return { ok: true, version: nuevaVersion };
  } catch(e) {
    Logger.log('❌ Error forzarResyncUsuarios: ' + e.message);
    // Si no hay UI (llamado desde el web app, no desde Sheets), no falla
    return { ok: false, message: e.message };
  }
}

// Abrevia el nombre completo para ahorrar caracteres en el reporte WhatsApp.
// Regla: conserva todos los nombres (uno o dos) + el PRIMER apellido. Omite el
// segundo apellido. Ejemplos:
//   "Víctor Alejandro Medina Sámano"  → "Víctor Alejandro Medina"
//   "Jairo Kayam Nava Mejía"          → "Jairo Kayam Nava"
//   "Eric Suaents Álvarez"            → "Eric Suaents"
//   "Brayan Colorado Torres"          → "Brayan Colorado"
//   "Marissa Aida Aguilar Salgado"    → "Marissa Aida Aguilar"
//
// Heurística: una persona típicamente tiene 1-2 nombres y 1-2 apellidos. Si el
// nombre tiene 4+ partes, asumimos los primeros (length-2) son nombres y los
// últimos 2 son apellidos → omitimos el último (apellido2). Si tiene 3 partes
// asumimos 1 nombre + 2 apellidos → omitimos el último. Si tiene ≤2 partes, no
// hay nada que cortar.
function abreviarNombre(nombreCompleto) {
  if (!nombreCompleto) return '';
  const partes = nombreCompleto.toString().trim().split(/\s+/);
  if (partes.length <= 2) return partes.join(' ');
  if (partes.length === 3) return partes.slice(0, 2).join(' ');           // omite apellido2
  return partes.slice(0, partes.length - 1).join(' ');                     // omite la última palabra
}

// Helper compartido para limpiar nombres y matchear contra deptoPorNombre
// (mismo enfoque que en el reporte WhatsApp principal).
function _limpiarNombreParaDepto(s) {
  if (!s) return '';
  return s.toString()
    .replace(/[\u00A0\u200B\u200C\u200D\uFEFF]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}
function _normParaDepto(s) {
  return _limpiarNombreParaDepto(s).toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
}

function _resolverDepto(nombre, deptoPorNombre) {
  if (!nombre) return 'SIN DEPTO';
  const nLimpio = _limpiarNombreParaDepto(nombre);
  if (deptoPorNombre[nLimpio]) return deptoPorNombre[nLimpio];
  // Fallback: matching normalizado contra todos los keys
  const nNorm = _normParaDepto(nombre);
  const fallback = Object.keys(deptoPorNombre).find(k => _normParaDepto(k) === nNorm);
  return fallback ? deptoPorNombre[fallback] : 'SIN DEPTO';
}

function agruparPorDepto(lista, deptoPorNombre) {
  const grupos = {};
  lista.forEach(nombre => {
    const depto = _resolverDepto(nombre, deptoPorNombre);
    if (!grupos[depto]) grupos[depto] = [];
    grupos[depto].push(abreviarNombre(nombre));
  });
  return grupos;
}

function renderGrupos(grupos) {
  let txt = '';
  Object.keys(grupos).sort().forEach(depto => {
    txt += '_' + depto + '_\n';
    grupos[depto].forEach(n => { txt += '• ' + n + '\n'; });
  });
  return txt;
}

function agruparPorDeptoConInfo(lista, deptoPorNombre, faltasMap, vacMap) {
  const grupos = {};
  lista.forEach(nombre => {
    const depto = _resolverDepto(nombre, deptoPorNombre);
    if (!grupos[depto]) grupos[depto] = [];

    let info = abreviarNombre(nombre);

    if (faltasMap && faltasMap[nombre]) {
      const f = faltasMap[nombre];
      if (typeof f === 'object' && 'total' in f) {
        // Formato compacto en UNA línea: "3 faltas (Q1:1, Q2:2)"
        const partes = [];
        if (f.q1 > 0) partes.push('Q1:' + f.q1);
        if (f.q2 > 0) partes.push('Q2:' + f.q2);
        const detalle = partes.length ? ' (' + partes.join(', ') + ')' : '';
        info += ' — ' + f.total + (f.total === 1 ? ' falta' : ' faltas') + detalle;
      } else {
        info += ' — ' + f + (f === 1 ? ' falta' : ' faltas');
      }
    }

    if (vacMap && vacMap[nombre]) {
      const v = vacMap[nombre];
      info += ' (día ' + v.diaActual + '/' + v.totalDias + ')';
    }

    grupos[depto].push(info);
  });
  return grupos;
}

    grupos[depto].forEach(n => { txt += '• ' + n + '\n'; });

}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var fnName = payload.fn;
    var args   = payload.args || [];

    var permitidas = [
      'validarPin','validarContrasena','guardarContrasena','verificarTieneContrasena',
      'getTodosLosUsuarios','getVersionUsuarios',
      'getSheetData','getConfigTurnos','getBonoPuntualidad','getChoferes','getZonasValidas',
      'guardarChecadaChofer','getActividadesIncidencias','inyectarIncidencia',
      'getIncidenciasReporte','getIncidenciasEmpleados','getEmpleadosConFechaContratacion',
      'getEnfermedades','inyectarEventualidades','inyectarDatosChecadores',
      'procesarAsistenciasCompleto','enviarReporteWhatsAppDesdeWeb',
      'getArchivosEnCarpeta','inyectarDesdeArchivoDrive',
      'guardarAvatarOverride','getAvatarOverrides','obtenerUltimosLogs','generarTextoReporte'
    ];

    if (permitidas.indexOf(fnName) === -1) {
      return _jsonResponse({ error: true, message: 'No permitida: ' + fnName });
    }

    var fn = eval(fnName);
    return _jsonResponse(fn.apply(null, args));

  } catch (err) {
    return _jsonResponse({ error: true, message: err.message });
  }
}

function _jsonResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

      ausentes.forEach(function(n) { msg += '• ' + n + '\n'; });

      msg += '\n🏖️ *VACACIONES:*\n';
      nombresVacaciones.forEach(function(n) { msg += '• ' + n + '\n'; });

  }
